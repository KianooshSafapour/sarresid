    81→/**
    82→ * Normalize a scanned/typed order code for matching:
    83→ * strip Code39 start/stop `*` and all whitespace, map Persian/Arabic digits
    84→ * to ASCII (manual entry), uppercase. (Inline equivalent of sanitizeCode39
    85→ * from lib/code39 — that one keeps spaces, which scanners do send, and its
    86→ * charset filter would mangle the code shown in error messages.)
    87→ */
    88→function normalizeOrderCode(raw: string): string {
    89→  return String(raw ?? '')
    90→    .replace(/[\s*]+/g, '')
    91→    .replace(/[۰-۹]/g, (d) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d)))
    92→    .replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
    93→    .toUpperCase()
    94→}
    95→
    96→/**
    97→ * Stable-read dedupe for the camera detection loop: a raw read only counts
    98→ * when the SAME value is seen twice in a row (two consecutive detector hits)
    99→ * — filters one-off glitches/ghost reads. Pure so QA can eval it directly.
   100→ */
   101→function dedupeRead(last: string, raw: string): { accept: boolean; last: string } {
   102→  if (!raw) return { accept: false, last: '' }
   103→  return { accept: raw === last, last: raw }
   104→}
   105→
   106→/* QA/debug handles (same convention as window.__errs) — lets headless QA eval
   107→ * the scan normalization/dedupe logic without a camera. */
   108→if (typeof window !== 'undefined') {
   109→  ;(window as unknown as Record<string, unknown>).__hzScanDebug = { normalizeOrderCode, dedupeRead }
   110→}

   112→/** numeric input → number | null (accepts Persian digits too) */
   113→function num(v: string): number | null {
   114→  const t = String(v).trim().replace(/[۰-۹]/g, (d) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d)))
   115→  if (!t) return null
   116→  const n = Number(t)
   117→  return isFinite(n) ? n : null
   118→}

   120→const round1 = (n: number) => Math.round(n * 100) / 100

   122→function initRows(mode: SheetMode, order: DetailOrder): Record<number, RowState> {
   123→  const rec: Record<number, RowState> = {}
   124→  for (const it of order.items) {
   125→    rec[it.id] = {
   126→      qty: mode === 'receive'
   127→        ? (it.deliveredQty ?? it.qty)
   128→        : (it.confirmedQty ?? it.deliveredQty ?? it.qty),
   129→      printed: String(it.printedPrice ?? it.unitCost ?? 0),
   130→      final: String(it.finalCost ?? it.printedPrice ?? it.unitCost ?? 0),
   131→      status: it.status === 'MISSING' || it.status === 'REJECTED' ? it.status : 'OK',
   132→      note: it.note ?? '',
   133→    }
   134→  }
   135→  return rec
   136→}

   185→function DeliverySheet({ orderId, mode, viaScan = false, user, onClose, onDone }: {
   186→  orderId: number | null
   187→  mode: SheetMode
   188→  /** sheet opened by an order-code scan (resolveScan hit) → focus the first
   189→   * qty input instead of the inner scan box + briefly flash the first row */
   190→  viaScan?: boolean
   191→  user: PUser
   192→  onClose: () => void
   193→  onDone: () => void
   194→}) {
   195→  const [order, setOrder] = React.useState<DetailOrder | null>(null)
   196→  const [loading, setLoading] = React.useState(false)
   197→  const [rows, setRows] = React.useState<Record<number, RowState>>({})
   198→  const [scan, setScan] = React.useState('')
   199→  const [saving, setSaving] = React.useState(false)
   200→  const [barcodeFor, setBarcodeFor] = React.useState<number | null>(null)
   201→  const [barcodeVal, setBarcodeVal] = React.useState('')
   202→  const [qtyFlash, setQtyFlash] = React.useState(false) // first-row gold ring flash (scan entry)
   203→  const firstQtyRef = React.useRef<HTMLInputElement | null>(null)
   204→  const qtyFocusDone = React.useRef<number | null>(null) // focus once per opened order
   205→
   206→  React.useEffect(() => {
   207→    if (orderId == null) { setOrder(null); return }
   208→    let alive = true
   209→    setLoading(true)
   210→    setOrder(null)
   211→    setScan(''); setBarcodeFor(null); setBarcodeVal('')
   212→    api.get<DetailOrder>(`/api/orders/${orderId}`)
   213→      .then((o) => { if (!alive) return; setOrder(o); setRows(initRows(mode, o)) })
   214→      .catch(() => { if (!alive) return; toast.error('خطا در بارگذاری سفارش'); onClose() })
   215→      .finally(() => { if (alive) setLoading(false) })
   216→    return () => { alive = false }
   217→  }, [orderId, mode, onClose])
   218→
   219→  /* scan-to-receive shortcut: when the sheet was opened by a code scan, put
   220→   * the caret straight into the first qty input (NOT the scan box) and briefly
   221→   * flash the first row so the operator lands where the counts are. */
   222→  React.useEffect(() => {
   223→    if (!viaScan || mode !== 'receive') return
   224→    if (loading || !order || order.items.length === 0) return
   225→    if (qtyFocusDone.current === order.id) return
   226→    qtyFocusDone.current = order.id
   227→    const raf = requestAnimationFrame(() => firstQtyRef.current?.focus())
   228→    setQtyFlash(true)
   229→    const timer = setTimeout(() => setQtyFlash(false), 1500)
   230→    return () => { cancelAnimationFrame(raf); clearTimeout(timer) }
   231→  }, [viaScan, mode, loading, order])
   232→
   233→  const upd = (id: number, patch: Partial<RowState>) =>
   234→    setRows((r) => (r[id] ? { ...r, [id]: { ...r[id], ...patch } } : r))
   235→
   236→  /* ---- barcode scan: mark item OK with full qty ---- */
   237→  const handleScan = () => {
   238→    const code = scan.trim()
   239→    if (!code || !order) return
   240→    const it = order.items.find((i) => (i.barcode ?? '') === code || (i.product?.barcode ?? '') === code)
   241→    if (!it) { toast.error('قلمی با این بارکد در سفارش یافت نشد'); return }
   242→    const q = mode === 'receive' ? it.qty : (it.deliveredQty ?? it.qty)
   243→    upd(it.id, { status: 'OK', qty: q, note: '' })
   244→    toast.success(`تایید شد ✓ — ${it.name}`)
   245→    setScan('')
   246→  }
   247→
   248→  /* ---- add barcode to an item (product) ---- */
   249→  const saveBarcode = async (itemId: number) => {
   250→    const barcode = barcodeVal.trim()
   251→    if (!barcode || orderId == null) return
   252→    try {
   253→      await api.patch(`/api/orders/${orderId}`, { action: 'addBarcode', userId: user.id, itemId, barcode })
   254→      toast.success('بارکد ذخیره شد ✓')
   255→      setBarcodeVal('')
   256→      setBarcodeFor(null)
   257→      setOrder((o) => (o
   258→        ? { ...o, items: o.items.map((i) => (i.id === itemId && !i.barcode ? { ...i, barcode } : i)) }
   259→        : o))
   260→    } catch (e) {
   261→      toast.error(e instanceof Error ? e.message : 'خطا در ذخیره بارکد')
   262→    }
   263→  }
   264→
   265→  /* ---- recalc footer ---- */
   266→  const totals = React.useMemo(() => {
   267→    if (!order) return { subtotal: 0, vat: 0, total: 0 }
   268→    let subtotal = 0
   269→    for (const it of order.items) {
   270→      const row = rows[it.id]
   271→      if (!row) continue
   272→      const cost = mode === 'receive'
   273→        ? (num(row.printed) ?? it.unitCost)
   274→        : (num(row.final) ?? it.printedPrice ?? it.unitCost)
   275→      subtotal += cost * row.qty
   276→    }
   277→    const vat = subtotal * VAT_RATE
   278→    return { subtotal, vat, total: subtotal + vat }
   279→  }, [order, rows, mode])
   280→
   281→  const delta = order ? totals.total - order.total : 0
   282→  const smallDelta = Math.abs(delta) <= DELTA_TOLERANCE
   283→
   284→  const submit = async () => {
   285→    if (!order) return
   286→    setSaving(true)
   287→    try {
   288→      const items = order.items.map((it) => {
   289→        const row = rows[it.id] ?? { qty: 0, printed: '', final: '', status: 'OK' as ItemStatus, note: '' }
   290→        return mode === 'receive'
   291→          ? {
   292→            id: it.id,
   293→            deliveredQty: row.qty,
   294→            printedPrice: num(row.printed),
   295→            status: row.status,
   296→            note: row.note || null,
   297→            finalCost: num(row.printed),
   298→          }
   299→          : {
   300→            id: it.id,
   301→            confirmedQty: row.qty,
   302→            finalCost: num(row.final),
   303→            status: row.status,
   304→            note: row.note || null,
   305→          }
   306→      })
   307→      await api.patch(`/api/orders/${order.id}`, {
   308→        action: mode === 'receive' ? 'receive' : 'confirm',
   309→        userId: user.id,
   310→        items,
   311→      })
   312→      toast.success(mode === 'receive' ? 'تحویل سفارش ثبت شد ✓' : 'تایید نهایی انبار ثبت شد ✓')
   313→      onDone()
   314→      onClose()
   315→    } catch (e) {
   316→      toast.error(e instanceof Error ? e.message : 'خطا در ثبت')
   317→    } finally {
   318→      setSaving(false)
   319→    }
   320→  }
   321→
   322→  const isToday = order ? order.receivingDate.slice(0, 10) === todayISO() : false
   323→
   324→  return (
   325→    <Modal
   326→      open={orderId != null}
   327→      onClose={onClose}
   328→      wide
   329→      title={mode === 'receive' ? 'دریافت تحویل سفارش' : 'تایید نهایی انبار'}
   330→    >
   331→      {loading || !order ? (
   332→        <Loading label="در حال بارگذاری اقلام سفارش…" />
   333→      ) : (
   334→        <div className="space-y-4">
   335→          {/* ---- header ---- */}
   336→          <div className="flex flex-wrap items-center justify-between gap-2 rounded-2xl border border-[#E4DCC8] bg-white p-4">
   337→            <div>
   338→              <div className="flex items-center gap-2">
   339→                <span className="text-lg font-extrabold text-[#253A2A]">{order.code}</span>
   340→                <StatusBadge status={order.status} />
   341→              </div>
   342→              <div className="mt-0.5 text-sm text-[#6B7A66]">تامین‌کننده: <b className="text-[#3E6B4A]">{order.supplier?.name ?? '—'}</b></div>
   343→            </div>
   344→            <div className="text-left">
   345→              <Badge className={isToday ? 'border-emerald-200 bg-emerald-50 text-emerald-700' : 'border-[#E4DCC8] bg-[#F5F2E8] text-[#4A5A44]'}>
   346→                {isToday ? 'تحویل صبح امروز' : `تحویل: ${fmtJalali(order.receivingDate)}`}
   347→              </Badge>
   348→              <div className="mt-1 text-xs text-[#8A9884]">
   349→                جمع اصلی سفارش: <b className="tabular-nums">{fmtMoney(order.total)}</b> · {toFaDigits(order.items.length)} قلم
   350→              </div>
   351→            </div>
   352→          </div>
   353→
   354→          {/* ---- scan box ---- */}
   355→          <div className="flex items-center gap-2 rounded-xl border border-[#93C572]/50 bg-[#F3F7EF] px-3 py-1.5">
   356→            <ScanLine size={18} className="shrink-0 text-[#3E6B4A]" />
   357→            <input
   358→              autoFocus={!viaScan}
   359→              value={scan}
   360→              onChange={(e) => setScan(e.target.value)}
   361→              onKeyDown={(e) => { if (e.key === 'Enter') { e.preventDefault(); handleScan() } }}
   362→              placeholder="بارکد را اسکن کنید و Enter بزنید…"
   363→              dir="ltr"
   364→              className="h-11 w-full bg-transparent text-sm text-[#253A2A] outline-none placeholder:text-[#A8A28C]"
   365→              aria-label="اسکن بارکد"
   366→            />
   367→          </div>
   368→
   369→          {/* ---- item cards ---- */}
   370→          <div className="max-h-[46vh] space-y-3 overflow-y-auto pl-1 pz-scroll">
   371→            {order.items.map((it, idx) => {
   372→              const row = rows[it.id]
   373→              if (!row) return null
   374→              const cost = mode === 'receive'
   375→                ? (num(row.printed) ?? it.unitCost)
   376→                : (num(row.final) ?? it.printedPrice ?? it.unitCost)
   377→              const line = cost * row.qty
   378→              return (
   379→                <Card key={it.id} className={cn('p-3', idx === 0 && qtyFlash && 'pz-focus-flash')}>
   380→                  {/* product head */}
   381→                  <div className="flex items-center gap-3">
   382→                    <ProductImage src={it.product?.imageUrl} name={it.name} size={48} />
   383→                    <div className="min-w-0 flex-1">
   384→                      <div className="truncate text-sm font-bold text-[#253A2A]">{it.name}</div>
   385→                      <div className="font-mono text-[11px] text-[#8A9884]" dir="ltr">{it.barcode || it.product?.barcode || '—'}</div>
   386→                      <div className="text-[11px] text-[#6B7A66]">
   387→                        مقدار سفارش: <b className="tabular-nums">{toFaDigits(it.qty)}</b>
   388→                        {mode === 'receive' && it.deliveredQty != null && <> · دریافتی قبلی: <b className="tabular-nums">{toFaDigits(it.deliveredQty)}</b></>}
   389→                      </div>
   390→                    </div>
   391→                    <div className="shrink-0 text-left">
   392→                      <div className="text-[10px] text-[#8A9884]">جمع خط</div>
   393→                      <div className="text-xs font-bold tabular-nums text-[#3E6B4A]">{fmtMoney(line)}</div>

   395→                  </div>
   396→
   397→                  {/* qty + price */}
   398→                  <div className="mt-3 grid gap-3 sm:grid-cols-2">
   399→                    <QtyStepper
   400→                      label={mode === 'receive' ? 'تعداد دریافتی' : 'تعداد تایید نهایی'}
   401→                      value={row.qty}
   402→                      onChange={(n) => upd(it.id, { qty: n })}
   403→                      inputRef={idx === 0 ? firstQtyRef : undefined}
   404→                    />
   405→                    {mode === 'receive' ? (
   406→                      <Field label="قیمت چاپ‌شده (تومان)" hint="قیمت چاپ‌شده روی بسته">
   407→                        <input
   408→                          type="number" inputMode="decimal" min={0} step="any"
   409→                          value={row.printed}
   410→                          onChange={(e) => upd(it.id, { printed: e.target.value })}
   411→                          className={inputCls}
   412→                        />
   413→                      </Field>
   414→                    ) : (
   415→                      <Field label="قیمت نهایی واحد (تومان)" hint="پیش‌فرض: قیمت چاپ‌شده / قیمت خرید">
   416→                        <input
   417→                          type="number" inputMode="decimal" min={0} step="any"
   418→                          value={row.final}
   419→                          onChange={(e) => upd(it.id, { final: e.target.value })}
   420→                          className={inputCls}
   421→                        />
   422→                      </Field>
   423→                    )}
   424→                  </div>
   425→
   426→                  {/* status toggles */}
   427→                  <div className="mt-3 grid grid-cols-3 gap-2">
   428→                    {STATUS_OPTS.map((o) => (
   429→                      <button
   430→                        key={o.v} type="button"

   434→                          'min-h-[44px] rounded-xl border-2 px-2 text-sm font-bold transition-all active:scale-[0.98]',

   441→                  </div>

   443→                  {/* note when MISSING / REJECTED */}
   444→                  {row.status !== 'OK' && (
   445→                    <div className="mt-3">
   446→                      <Field label="توضیح" hint="دلیل ناقص یا مردود بودن را بنویسید">

   448→                          value={row.note}
   449→                          onChange={(e) => upd(it.id, { note: e.target.value })}
   450→                          placeholder="مثلاً: ۲ عدد شکسته بود"
   451→                          className={inputCls}

  1022→  onStatusChange?: (status: CamStatus) => void

  1143→    document.addEventListener('visibilitychange', onVis)

  1146→      cancelled = true
  1147→      if (timer) { clearInterval(timer); timer = null }
  1148→      document.removeEventListener('keydown', onKey)
  1149→      document.removeEventListener('visibilitychange', onVis)
  1150→      stopStream()
  1151→    }
  1152→  }, [open, onClose, onCode, onStatusChange, stopStream])
  1153→
  1154→  if (!open) return null
  1155→
  1156→  const fallback = status === 'unsupported' || status === 'unavailable'
  1157→
  1158→  return (
  1159→    <div
  1160→      dir="rtl"
  1161→      role="dialog"
  1162→      aria-modal="true"
  1163→      aria-label="اسکن بارکد سفارش با دوربین"

  1166→      {/* header */}
  1167→      <div className="flex items-center justify-between gap-2 px-4 pt-4">
  1168→        <div className="flex items-center gap-2 text-[#F3E7C4]">
  1169→          <Camera size={18} />
  1170→          <span className="text-sm font-bold">اسکن با دوربین</span>

  1176→          className="flex h-11 w-11 items-center justify-center rounded-full border border-[#DAA520]/40 bg-black/30 text-[#F3E7C4] transition hover:bg-black/50 active:scale-95"
  1177→        >
  1178→          <X size={20} />

  1185→          <div className="mx-auto flex w-full max-w-sm flex-col items-center gap-4 rounded-3xl border border-[#DAA520]/30 bg-[#1B221A]/90 p-6 text-center shadow-2xl">
  1186→            <span className="flex h-14 w-14 items-center justify-center rounded-2xl border border-rose-400/30 bg-rose-500/10 text-rose-300">
  1187→              <CameraOff size={26} />
  1188→            </span>
  1189→            <div className="text-base font-extrabold text-[#F3E7C4]">
  1190→              {status === 'unsupported' ? 'مرورگر شما از اسکن دوربینی پشتیبانی نمی‌کند' : 'دسترسی به دوربین ممکن نشد'}
