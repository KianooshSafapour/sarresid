    13→/* ---------- local types ---------- */
    14→type CompanyMini = { id: number; name: string }
    15→type CompanyProduct = { id: number; name: string; stock: number; minStock: number; sellPrice: number; imageUrl: string | null; active: boolean }
    16→type CompanyRow = {
    17→  id: number
    18→  name: string
    19→  note?: string | null
    20→  products: CompanyProduct[]
    21→  suppliers?: { supplier: { id: number; name: string } }[]
    22→}
    23→type SupplierRow = SupplierT & {
    24→  companies?: { company: CompanyMini }[]
    25→  _count?: { products: number }
    26→}
    27→
    28→const KIND_META: Record<string, { label: string; cls: string }> = {
    29→  MANUFACTURER: { label: 'تولیدکننده', cls: 'border-[#C8D8C0] bg-[#F3F7EF] text-[#3E6B4A]' },
    30→  DISTRIBUTOR: { label: 'پخش', cls: 'border-[#EAD9A8] bg-[#FBF3DC] text-[#8A6508]' },
    31→  BOTH: { label: 'هردو', cls: 'border-[#D9C9A8] bg-[#F5EFE0] text-[#6B5B2A]' },
    32→}
    33→const PAY_LABEL: Record<string, string> = { CASH: 'نقدی', CHEQUE: 'چک', MIXED: 'مختلط' }
    34→const PAY_CLS: Record<string, string> = {
    35→  CASH: 'border-emerald-200 bg-emerald-50 text-emerald-700',
    36→  CHEQUE: 'border-amber-200 bg-amber-50 text-amber-700',
    37→  MIXED: 'border-stone-200 bg-stone-100 text-stone-600',
    38→}
    39→
    40→function KindBadge({ kind }: { kind: string }) {
    41→  const m = KIND_META[kind] ?? { label: kind, cls: 'border-stone-200 bg-stone-100 text-stone-600' }
    42→  return <Badge className={m.cls}>{m.label}</Badge>
    43→}
    44→
    45→function PayChip({ terms, chequeDays }: { terms: string; chequeDays: number }) {
    46→  return (
    47→    <Badge className={PAY_CLS[terms] ?? PAY_CLS.MIXED}>
    48→      {PAY_LABEL[terms] ?? terms}
    49→      {terms !== 'CASH' && chequeDays > 0 ? ` · ${toFaDigits(chequeDays)} روزه` : ''}
    50→    </Badge>
    51→  )
    52→}
    53→
    54→function CompanyChip({ name }: { name: string }) {
    55→  return (
    56→    <span className="inline-flex max-w-36 items-center gap-1.5 rounded-full border border-[#E4DCC8] bg-white py-0.5 pl-2.5 pr-0.5 text-xs font-medium text-[#4A5A44]">
    57→      <span className="flex h-5 w-5 shrink-0 items-center justify-center rounded-full bg-gradient-to-br from-[#3E6B4A] to-[#5F8F55] text-[10px] font-bold text-white" title={name}>
    58→        {name.trim().charAt(0).toUpperCase()}
    59→      </span>
    60→      <span className="truncate">{name}</span>
    61→    </span>
    62→  )
    63→}
    64→
    65→function Crumb({ label, onClick, active }: { label: string; onClick?: () => void; active?: boolean }) {
    66→  return (
    67→    <button
    68→      type="button"
    69→      onClick={onClick}
    70→      disabled={active}
    71→      className={`rounded-lg px-2 py-1 text-sm font-semibold transition ${active ? 'cursor-default text-[#253A2A]' : 'text-[#5F8F55] hover:bg-[#F3F7EF]'}`}
    72→    >
    73→      {label}
    74→    </button>
    75→  )
    76→}
    77→
    78→/** product mini-card — same look as ProductsSection (stock number at the bottom, colored) */
    79→function ProductMiniCard({ p }: { p: CompanyProduct }) {
    80→  return (
    81→    <Card className={`flex flex-col gap-2 p-3 ${p.active ? '' : 'opacity-60'}`}>
    82→      <div className="flex items-start justify-between gap-2">
    83→        <ProductImage src={p.imageUrl} name={p.name} size={44} />
    84→        <StockBadge stock={p.stock} minStock={p.minStock} />
    85→      </div>
    86→      <div className="truncate text-xs font-bold text-[#253A2A]" title={p.name}>{p.name}</div>
    87→      <Money value={p.sellPrice} className="text-xs font-bold text-[#8A6508]" />
    88→      <div className="mt-auto flex items-end justify-between border-t border-[#EFEAD8] pt-1.5">
    89→        <span className={`text-2xl font-black leading-none tabular-nums ${stockDot(p.stock, p.minStock)}`}>
    90→          {toFaDigits(Math.floor(p.stock))}
    91→        </span>
    92→        <span className="text-[10px] font-medium text-[#8A9884]">موجودی</span>
    93→      </div>
    94→    </Card>
    95→  )
    96→}

    98→export default function SuppliersSection({ user }: { user: PUser }) {
    99→  const manage =
   100→    hasRole(user, 'GENERAL_MANAGER') || hasRole(user, 'PRODUCT_MANAGER') ||
   101→    hasRole(user, 'OPERATION_MANAGER') || hasRole(user, 'IT_ADMIN')
   102→  // price-list XLS export — same gate as the API (plus PRODUCT_MANAGER)
   103→  const canExportPrices =
   104→    hasRole(user, 'ACCOUNTANT') || hasRole(user, 'GENERAL_MANAGER') ||
   105→    hasRole(user, 'OWNER') || hasRole(user, 'IT_ADMIN') || hasRole(user, 'PRODUCT_MANAGER')
   106→
   107→  const [suppliers, setSuppliers] = React.useState<SupplierRow[]>([])
   108→  const [companies, setCompanies] = React.useState<CompanyRow[]>([])
   109→  const [priceImport, setPriceImport] = React.useState<PriceImportInfo | null>(null)
   110→  const [loading, setLoading] = React.useState(true)
   111→  const [search, setSearch] = React.useState('')
   112→
   113→  /* drill-down: list → supplier → company */
   114→  const [openSupplier, setOpenSupplier] = React.useState<SupplierRow | null>(null)
   115→  const [openCompany, setOpenCompany] = React.useState<CompanyRow | null>(null)
   116→
   117→  const load = React.useCallback(async () => {
   118→    setLoading(true)
   119→    try {
   120→      const [s, c] = await Promise.all([
   121→        api.get<{ suppliers: SupplierRow[]; priceImport?: PriceImportInfo | null }>('/api/suppliers'),
   122→        api.get<{ companies: CompanyRow[] }>('/api/companies'),
   123→      ])
   124→      setSuppliers(s.suppliers ?? [])
   125→      setPriceImport(s.priceImport ?? null)
   126→      setCompanies(c.companies ?? [])
   127→    } catch (e) {
   128→      toast.error(e instanceof Error ? e.message : 'خطا در دریافت تامین‌کنندگان')
   129→    } finally {
   130→      setLoading(false)
   131→    }
   132→  }, [])
   133→
   134→  React.useEffect(() => { void load() }, [load])
   135→
   136→  /* ---------- supplier modal ---------- */
   137→  const [supOpen, setSupOpen] = React.useState(false)
   138→  const [editingSup, setEditingSup] = React.useState<SupplierRow | null>(null)
   139→  const [supForm, setSupForm] = React.useState({
   140→    name: '', kind: 'DISTRIBUTOR', phone: '',
   141→    paymentTerms: 'CASH', chequeDays: '30', note: '', companyIds: [] as number[],
   142→  })
   143→  const [savingSup, setSavingSup] = React.useState(false)
   144→
   145→  const openSupplierModal = (s: SupplierRow | null) => {
   146→    setEditingSup(s)
   147→    setSupForm({
   148→      name: s?.name ?? '',
   149→      kind: s?.kind ?? 'DISTRIBUTOR',
   150→      phone: s?.phone ?? '',
   151→      paymentTerms: s?.paymentTerms ?? 'CASH',
   152→      chequeDays: String(s?.chequeDays ?? 30),
   153→      note: s?.note ?? '',
   154→      companyIds: (s?.companies ?? []).map((c) => c.company.id),
   155→    })
   156→    setSupOpen(true)
   157→  }
   158→
   159→  const toggleCompany = (id: number) =>
   160→    setSupForm((f) => ({
   161→      ...f,
   162→      companyIds: f.companyIds.includes(id) ? f.companyIds.filter((x) => x !== id) : [...f.companyIds, id],
   163→    }))
   164→
   165→  const saveSupplier = async () => {
   166→    if (!supForm.name.trim()) { toast.error('نام تامین‌کننده الزامی است'); return }
   167→    setSavingSup(true)
   168→    try {
   169→      const payload = {
   170→        name: supForm.name.trim(),
   171→        kind: supForm.kind,
   172→        phone: supForm.phone.trim() || null,
   173→        note: supForm.note.trim() || null,
   174→        paymentTerms: supForm.paymentTerms,
   175→        chequeDays: Number(supForm.chequeDays) || 0,
   176→        companyIds: supForm.companyIds,
   177→        userId: user.id,
   178→        userName: user.name,
   179→      }
   180→      if (editingSup?.id) {
   181→        await api.patch('/api/suppliers', { id: editingSup.id, ...payload })
   182→        toast.success('تامین‌کننده بروزرسانی شد')
   183→      } else {
   184→        await api.post('/api/suppliers', payload)
   185→        toast.success('تامین‌کننده جدید ثبت شد')
   186→      }
   187→      setSupOpen(false)
   188→      void load()
   189→    } catch (e) {
   190→      toast.error(e instanceof Error ? e.message : 'خطا در ذخیره تامین‌کننده')
   191→    } finally {
   192→      setSavingSup(false)
   193→    }
   194→  }
   195→
   196→  /* ---------- company modal ---------- */
   197→  const [comOpen, setComOpen] = React.useState(false)
   198→  const [comName, setComName] = React.useState('')
   199→  const [savingCom, setSavingCom] = React.useState(false)
   200→
   201→  const saveCompany = async () => {
   202→    if (!comName.trim()) { toast.error('نام شرکت الزامی است'); return }
   203→    setSavingCom(true)
   204→    try {
   205→      await api.post('/api/companies', { name: comName.trim(), userId: user.id, userName: user.name })
   206→      toast.success('شرکت جدید ثبت شد')
   207→      setComOpen(false)
   208→      setComName('')
   209→      void load()
   210→    } catch (e) {
   211→      toast.error(e instanceof Error ? e.message : 'خطا در ثبت شرکت')
   212→    } finally {
   213→      setSavingCom(false)
   214→    }
   215→  }
   216→
   217→  /* ---------- derived ---------- */
   218→  const PRICE_STALE_MS = 30 * 24 * 3600 * 1000
   219→  const priceStale = !priceImport || (Date.now() - new Date(priceImport.at).getTime()) > PRICE_STALE_MS
   220→
   221→  /* ---------- per-supplier price-list export (gold, round-trips with the import) ---------- */
   222→  const [exportingId, setExportingId] = React.useState<number | null>(null)
   223→  const exportPriceList = React.useCallback(async (s: SupplierRow) => {
   224→    setExportingId(s.id)
   225→    try {
   226→      const j = isoToJalali(new Date())
   227→      const slug = s.name.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '').slice(0, 40).replace(/-+$/g, '') || `supplier-${s.id}`
   228→      await downloadFile(`/api/export/price-list?supplierId=${s.id}&userId=${user.id}`, `price-list-${slug}-${j.jy}-${j.jm}.xls`)
   229→      toast.success(`فهرست قیمت ${s.name} دانلود شد ✓`)
   230→    } catch (e) {
   231→      toast.error(e instanceof Error && e.message !== 'Download failed' ? e.message : 'خطا در دریافت فهرست قیمت')
   232→    } finally {
   233→      setExportingId(null)
   234→    }
   235→  }, [user.id])
   236→
   237→  const q = search.trim().toLowerCase()
   238→  const filteredSuppliers = q
   239→    ? suppliers.filter((s) => s.name.toLowerCase().includes(q) || (s.companies ?? []).some((c) => c.company.name.toLowerCase().includes(q)))
   240→    : suppliers
   241→
   242→  const supplierCompanies = openSupplier
   243→    ? (openSupplier.companies ?? []).map((c) => companies.find((x) => x.id === c.company.id)).filter((x): x is CompanyRow => !!x)
   244→    : []
   245→
   246→  /* ---------- render: company products ---------- */
   247→  if (openSupplier && openCompany) {
   248→    const lowCount = openCompany.products.filter((p) => p.stock <= p.minStock).length
   249→    return (
   250→      <div>
   251→        <SectionHeader
   252→          title={openCompany.name}
   253→          subtitle={`محصولات شرکت ${openCompany.name} — ${toFaDigits(openCompany.products.length)} کالا${lowCount ? ` · ${toFaDigits(lowCount)} کمبود` : ''}`}
   254→          icon={<Boxes className="h-5 w-5" />}
   255→          actions={
   256→            <GhostButton onClick={() => setOpenCompany(null)} className="min-h-[44px]">
   257→              <ArrowRight className="h-4 w-4" /> بازگشت به {openSupplier.name}
   258→            </GhostButton>
   259→          }
   260→        />
   261→        <div className="mb-4 flex flex-wrap items-center gap-1 rounded-xl border border-[#E4DCC8] bg-white/80 px-2 py-1.5">
   262→          <Crumb label="همه تامین‌کنندگان" onClick={() => { setOpenSupplier(null); setOpenCompany(null) }} />
   263→          <ChevronLeft className="h-4 w-4 text-[#B8B29A]" />
   264→          <Crumb label={openSupplier.name} onClick={() => setOpenCompany(null)} />
   265→          <ChevronLeft className="h-4 w-4 text-[#B8B29A]" />
   266→          <Crumb label={openCompany.name} active />
   267→        </div>
   268→        {openCompany.products.length === 0 ? (
   269→          <EmptyState icon={<Boxes className="h-10 w-10" />} title="این شرکت هنوز کالایی ندارد" />
   270→        ) : (
   271→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 sm:gap-4 lg:grid-cols-4">
   272→            {openCompany.products.map((p) => <ProductMiniCard key={p.id} p={p} />)}
   273→          </div>
   274→        )}
   275→      </div>
   276→    )
   277→  }
   278→
   279→  /* ---------- render: supplier companies ---------- */
   280→  if (openSupplier) {
   281→    return (
   282→      <div>
   283→        <SectionHeader
   284→          title={openSupplier.name}
   285→          subtitle="برای دیدن محصولات، شرکت را انتخاب کنید"
   286→          icon={<Truck className="h-5 w-5" />}
   287→          actions={
   288→            <div className="flex gap-2">
   289→              {manage && (
   290→                <GhostButton onClick={() => openSupplierModal(openSupplier)} className="min-h-[44px]">
   291→                  <Pencil className="h-4 w-4" /> ویرایش تامین‌کننده
   292→                </GhostButton>
   293→              )}
   294→              <GhostButton onClick={() => setOpenSupplier(null)} className="min-h-[44px]">
   295→                <ArrowRight className="h-4 w-4" /> همه تامین‌کنندگان
   296→              </GhostButton>
   297→            </div>
   298→          }
   299→        />
   300→        <div className="mb-4 flex flex-wrap items-center gap-1 rounded-xl border border-[#E4DCC8] bg-white/80 px-2 py-1.5">
   301→          <Crumb label="همه تامین‌کنندگان" onClick={() => setOpenSupplier(null)} />
   302→          <ChevronLeft className="h-4 w-4 text-[#B8B29A]" />
   303→          <Crumb label={openSupplier.name} active />
   304→        </div>
   305→        {supplierCompanies.length === 0 ? (
   306→          <EmptyState icon={<Building2 className="h-10 w-10" />} title="شرکتی به این تامین‌کننده متصل نیست" hint="از دکمه ویرایش تامین‌کننده می‌توانید شرکت‌ها را متصل کنید." />
   307→        ) : (
   308→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3">
   309→            {supplierCompanies.map((c) => {
   310→              const low = c.products.filter((p) => p.stock <= p.minStock).length
   311→              return (
   312→                <Card
   313→                  key={c.id}
   314→                  onClick={() => setOpenCompany(c)}
   315→                  className="flex cursor-pointer items-center gap-3 p-4 transition hover:-translate-y-0.5 hover:border-[#5F8F55] hover:shadow-lg"
   316→                >
   317→                  <span className="flex h-11 w-11 shrink-0 items-center justify-center rounded-xl bg-gradient-to-br from-[#3E6B4A] to-[#5F8F55] text-lg font-bold text-white" title={c.name}>
   318→                    {c.name.trim().charAt(0).toUpperCase()}
   319→                  </span>
   320→                  <div className="min-w-0 flex-1">
   321→                    <div className="truncate text-sm font-bold text-[#253A2A]">{c.name}</div>
   322→                    <div className="mt-0.5 flex flex-wrap gap-1.5 text-[11px]">
   323→                      <Badge className="border-[#E4DCC8] bg-white text-[#4A5A44]">{toFaDigits(c.products.length)} محصول</Badge>
   324→                      {low > 0 && <Badge className="border-amber-200 bg-amber-50 text-amber-700">{toFaDigits(low)} کمبود</Badge>}
   325→                    </div>
   326→                  </div>
   327→                  <ChevronLeft className="h-5 w-5 shrink-0 text-[#B8B29A]" />
   328→                </Card>
   329→              )
   330→            })}
   331→          </div>
   332→        )}
   333→      </div>
   334→    )
   335→  }
   336→
   337→  /* ---------- render: list ---------- */
   338→  return (
   339→    <div>
   340→      <SectionHeader
   341→        title="تامین‌کنندگان | Suppliers"
   342→        subtitle="تولیدی‌ها و پخش‌ها، شرکت‌ها و محصولات هرکدام"
   343→        icon={<Truck className="h-5 w-5" />}
   344→        actions={
   345→          manage ? (
   346→            <>
   347→              <GhostButton onClick={() => setComOpen(true)} className="min-h-[44px]">
   348→                <Plus className="h-4 w-4" /> شرکت جدید
   349→              </GhostButton>
   350→              <PrimaryButton onClick={() => openSupplierModal(null)} className="min-h-[44px]">
   351→                <Plus className="h-4 w-4" /> تامین‌کننده جدید
   352→              </PrimaryButton>
   353→            </>
   354→          ) : undefined
   355→        }
   356→      />
   357→
   358→      {/* price-list recency chip — amber when the last price import is older than 30 days (or never) */}
   359→      {!loading && (
   360→        <div className="mb-4 flex flex-wrap items-center gap-2" data-price-recency data-price-stale={priceStale ? '1' : '0'}>
   361→          <Badge
   362→            className={priceStale
   363→              ? 'border-amber-300 bg-amber-50 px-3 py-1.5 text-amber-800'
   364→              : 'border-[#C8D8C0] bg-[#F3F7EF] px-3 py-1.5 text-[#3E6B4A]'}
   365→          >
   366→            <History className="h-3.5 w-3.5 shrink-0" />
   367→            <span className="font-semibold">
   368→              فهرست قیمت‌ها: {priceImport ? (
   369→                <>آخرین ورود <TimeAgo iso={priceImport.at} /> — {priceImport.by}</>
   370→              ) : (
   371→                'تاکنون وارد نشده'
   372→              )}
   373→            </span>
   374→            {priceStale && (
   375→              <span className="font-bold">— ورود فهرست قیمت | Import price list</span>
   376→            )}
   377→          </Badge>
   378→        </div>
   379→      )}
   380→
   381→      <div className="relative mb-4">
   382→        <Search className="absolute right-3 top-1/2 h-4 w-4 -translate-y-1/2 text-[#A8A28C]" />
   383→        <input
   384→          value={search}
   385→          onChange={(e) => setSearch(e.target.value)}
   386→          placeholder="جستجوی تامین‌کننده یا شرکت…"
   387→          className={`${inputCls} min-h-[44px] pr-9`}
   388→          aria-label="جستجوی تامین‌کننده"
   389→        />
   390→      </div>
   391→
   392→      {loading ? (
   393→        <Loading label="در حال دریافت تامین‌کنندگان…" />
   394→      ) : filteredSuppliers.length === 0 ? (
   395→        <EmptyState icon={<Truck className="h-10 w-10" />} title="تامین‌کننده‌ای پیدا نشد" />
   396→      ) : (
   397→        <div className="grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-3">
   398→          {filteredSuppliers.map((s) => {
   399→            const lastImport = s.lastPriceImport ?? null
   400→            const importStale = !lastImport || (Date.now() - new Date(lastImport.at).getTime()) > PRICE_STALE_MS
   401→            return (
   402→            <Card key={s.id} className="flex flex-col p-5">
   403→              <div className="flex items-start justify-between gap-2">
   404→                <button type="button" onClick={() => setOpenSupplier(s)} className="min-w-0 flex-1 text-right">
   405→                  <div className="truncate text-base font-bold text-[#253A2A]" title={s.name}>{s.name}</div>
   406→                  <div className="mt-2 flex flex-wrap gap-1.5">
   407→                    <KindBadge kind={s.kind} />
   408→                    <PayChip terms={s.paymentTerms} chequeDays={s.chequeDays} />
   409→                  </div>
   410→                </button>
   411→                {manage && (
   412→                  <button
   413→                    type="button"
   414→                    onClick={() => openSupplierModal(s)}
   415→                    className="flex h-9 w-9 shrink-0 items-center justify-center rounded-full border border-[#E4DCC8] bg-white text-[#4A5A44] transition hover:border-[#5F8F55] hover:text-[#3E6B4A]"
   416→                    title="ویرایش"
   417→                    aria-label={`ویرایش ${s.name}`}
   418→                  >
   419→                    <Pencil className="h-4 w-4" />
   420→                  </button>
   421→                )}
   422→              </div>
   423→
   424→              {s.phone && (
   425→                <div className="mt-3 flex items-center gap-2 text-xs text-[#4A5A44]" dir="ltr">
   426→                  <Phone className="h-3.5 w-3.5 text-[#8A9884]" /> <span className="tabular-nums">{s.phone}</span>
