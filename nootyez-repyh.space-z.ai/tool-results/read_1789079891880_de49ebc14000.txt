    75→function roleSubtitle(u: PUser): string {
    76→  if (hasRole(u, 'OWNER') || hasRole(u, 'GENERAL_MANAGER')) return 'نمای کلی مدیریت هایپر زیتون | Management overview'
    77→  if (hasRole(u, 'ACCOUNTANT')) return 'صف مالی، تسویه و چک‌ها | Financial queue'
    78→  if (hasRole(u, 'DELIVERY_RECEIVER') || hasRole(u, 'INVENTORY_SUPERVISOR')) return 'تحویل‌ها و موجودی امروز | Today’s deliveries'
    79→  if (hasRole(u, 'PRODUCT_MANAGER')) return 'سفارش‌ها و موجودی کالا | Orders & stock'
    80→  if (hasRole(u, 'OPERATION_MANAGER')) return 'عملیات روزانه فروشگاه | Daily store operations'
    81→  if (hasRole(u, 'IT_ADMIN')) return 'پایش سیستم و پشتیبانی | System & support'
    82→  return 'امروز چه کارهایی در جریان است | What’s happening today'
    83→}

   485→/* ================= Today Briefing (morning meeting card) ================= */
   486→
   487→interface BriefingData {
   488→  deliveries: { id: number; code: string; supplier: string; total: number; overdue: boolean }[]
   489→  cheques: { id: number; amount: number; dueDate: string; status: string }[]
   490→  urgentTasks: { id: number; title: string; priority: string; dueDate: string | null; mine: boolean }[]
   491→}
   492→
   493→function BriefingCard({ user }: { user: PUser }) {
   494→  const setView = useApp((s) => s.setView)
   495→  const openOrder = useApp((s) => s.openOrder)
   496→  const [data, setData] = React.useState<BriefingData | null>(null)
   497→
   498→  React.useEffect(() => {
   499→    let alive = true
   500→    Promise.all([
   501→      api.get<{ orders: { id: number; code: string; status: string; receivingDate: string; total: number; supplier?: { name: string } }[] }>('/api/orders?active=1'),
   502→      api.get<{ cheques: { id: number; amount: number; dueDate: string; status: string }[] }>('/api/cheques'),
   503→      api.get<{ tasks: { id: number; title: string; priority: string; status: string; dueDate: string | null; assignedToId: number }[] }>('/api/tasks'),
   504→    ])
   505→      .then(([o, ch, t]) => {
   506→        if (!alive) return
   507→        const today = todayISO()
   508→        const deliveries = o.orders
   509→          .filter((x) => ['APPROVED', 'SUBMITTED'].includes(x.status))
   510→          .map((x) => ({
   511→            id: x.id,
   512→            code: x.code,
   513→            supplier: x.supplier?.name ?? '—',
   514→            total: x.total,
   515→            overdue: daysBetweenISO(today, x.receivingDate.slice(0, 10)) < 0,
   516→          }))
   517→          .sort((a, b) => Number(b.overdue) - Number(a.overdue))
   518→          .slice(0, 4)
   519→        const cheques = ch.cheques
   520→          .filter((x) => ['APPROVED', 'WRITTEN', 'SIGNED', 'GIVEN'].includes(x.status))
   521→          .filter((x) => { const d = daysBetweenISO(today, x.dueDate.slice(0, 10)); return d >= -3 && d <= 7 })
   522→          .sort((a, b) => a.dueDate.localeCompare(b.dueDate))
   523→          .slice(0, 3)
   524→        const urgentTasks = t.tasks
   525→          .filter((x) => ['OPEN', 'IN_PROGRESS', 'PAUSED'].includes(x.status))
   526→          .filter((x) => x.priority === 'URGENT' || x.priority === 'HIGH' || x.assignedToId === user.id)
   527→          .slice(0, 4)
   528→          .map((x) => ({ id: x.id, title: x.title, priority: x.priority, dueDate: x.dueDate, mine: x.assignedToId === user.id }))
   529→        setData({ deliveries, cheques, urgentTasks })
   530→      })
   531→      .catch(() => { if (alive) setData({ deliveries: [], cheques: [], urgentTasks: [] }) })
   532→    return () => { alive = false }
   533→  }, [user.id])
   534→
   535→  const empty = !data || (data.deliveries.length === 0 && data.cheques.length === 0 && data.urgentTasks.length === 0)
   536→  if (empty) return null
   537→
   538→  return (
   539→    <div className="relative overflow-hidden rounded-2xl border border-[#EAD9A8] bg-gradient-to-l from-[#FBF3DC] via-[#FAF7EF] to-[#F3F7EF] p-4 shadow-sm sm:p-5">
   540→      <div className="mb-3 flex items-center gap-2">
   541→        <span className="flex h-8 w-8 items-center justify-center rounded-lg bg-gradient-to-br from-[#DAA520] to-[#B8860B] text-white shadow">
   542→          <Sunrise size={16} />
   543→        </span>
   544→        <div>
   545→          <h3 className="text-sm font-bold text-[#253A2A]">بریفینگ امروز | Today at a glance</h3>
   546→          <p className="text-[11px] text-[#8A9884]">خلاصه صبحگاهی برای جلسه ۲ دقیقه‌ای صبح</p>
   547→        </div>
   548→      </div>
   549→      <div className="grid gap-3 md:grid-cols-3">
   550→        {/* Deliveries — div (role=button) because it contains an inner real button */}
   551→        <div
   552→          role="button"
   553→          tabIndex={0}
   554→          onClick={() => setView('deliveries')}
   555→          onKeyDown={(e) => { if (e.key === 'Enter') setView('deliveries') }}
   556→          className="cursor-pointer rounded-xl border border-[#E4DCC8] bg-white/80 p-3 text-right transition hover:border-[#93C572] hover:shadow-sm"
   557→        >
   558→          <div className="mb-2 flex items-center justify-between">
   559→            <span className="text-xs font-bold text-[#4A5A44]">🚚 تحویل‌های در راه</span>
   560→            <ChevronLeft size={14} className="text-[#A8A28C]" />
   561→          </div>
   562→          {data!.deliveries.length === 0 ? (
   563→            <div className="text-[11px] text-[#8A9884]">امروز تحویلی در راه نیست</div>
   564→          ) : (
   565→            <div className="space-y-1.5">
   566→              {data!.deliveries.map((d) => (
   567→                <div key={d.id} className="flex items-center justify-between gap-2 text-[11px]">
   568→                  <button
   569→                    type="button"
   570→                    onClick={(e) => { e.stopPropagation(); openOrder(d.id) }}
   571→                    className="pz-barcode shrink-0 font-bold text-[#3E6B4A] hover:underline"
   572→                  >
   573→                    {d.code}
   574→                  </button>
   575→                  <span className="min-w-0 flex-1 truncate text-[#6B7A66]">{d.supplier}</span>
   576→                  {d.overdue && <span className="shrink-0 rounded-full bg-red-100 px-1.5 py-0.5 text-[9px] font-bold text-red-600">معوق!</span>}
