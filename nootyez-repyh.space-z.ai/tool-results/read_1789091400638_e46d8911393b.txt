    81→/**
    82→ * Normalize a scanned/typed order code for matching:
    83→ * strip Code39 start/stop `*` and all whitespace, map Persian/Arabic digits
    84→ * to ASCII (manual entry), uppercase. (Inline equivalent of sanitizeCode39
    85→ * from lib/code39 — that one keeps spaces, which scanners do send, and its
    86→ * charset filter would mangle the code shown in error messages.)
    87→ */
    88→function normalizeOrderCode(raw: string): string {
    89→  return String(raw ?? '')
    90→    .replace(/[\s*]+/g, '')
    91→    .replace(/[۰-۹]/g, (d) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d)))
    92→    .replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
    93→    .toUpperCase()
    94→}
    95→
    96→/**
    97→ * Stable-read dedupe for the camera detection loop: a raw read only counts
    98→ * when the SAME value is seen twice in a row (two consecutive detector hits)
    99→ * — filters one-off glitches/ghost reads. Pure so QA can eval it directly.
   100→ */
   101→function dedupeRead(last: string, raw: string): { accept: boolean; last: string } {
   102→  if (!raw) return { accept: false, last: '' }
   103→  return { accept: raw === last, last: raw }
   104→}
   105→
   106→/* QA/debug handles (same convention as window.__errs) — lets headless QA eval
   107→ * the scan normalization/dedupe logic without a camera. */
   108→if (typeof window !== 'undefined') {
   109→  ;(window as unknown as Record<string, unknown>).__hzScanDebug = { normalizeOrderCode, dedupeRead }
   110→}

   112→/** numeric input → number | null (accepts Persian digits too) */
   113→function num(v: string): number | null {
   114→  const t = String(v).trim().replace(/[۰-۹]/g, (d) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d)))
   115→  if (!t) return null
   116→  const n = Number(t)
   117→  return isFinite(n) ? n : null
   118→}
   119→
   120→const round1 = (n: number) => Math.round(n * 100) / 100

   122→function initRows(mode: SheetMode, order: DetailOrder): Record<number, RowState> {
   123→  const rec: Record<number, RowState> = {}
   124→  for (const it of order.items) {
   125→    rec[it.id] = {
   126→      qty: mode === 'receive'
   127→        ? (it.deliveredQty ?? it.qty)
   128→        : (it.confirmedQty ?? it.deliveredQty ?? it.qty),
   129→      printed: String(it.printedPrice ?? it.unitCost ?? 0),
   130→      final: String(it.finalCost ?? it.printedPrice ?? it.unitCost ?? 0),
   131→      status: it.status === 'MISSING' || it.status === 'REJECTED' ? it.status : 'OK',
   132→      note: it.note ?? '',
   133→    }
   134→  }
   135→  return rec
   136→}

   178→function DeliverySheet({ orderId, mode, user, onClose, onDone }: {
   179→  orderId: number | null
   180→  mode: SheetMode
   181→  user: PUser
   182→  onClose: () => void
   183→  onDone: () => void
   184→}) {
   185→  const [order, setOrder] = React.useState<DetailOrder | null>(null)
   186→  const [loading, setLoading] = React.useState(false)
   187→  const [rows, setRows] = React.useState<Record<number, RowState>>({})
   188→  const [scan, setScan] = React.useState('')
   189→  const [saving, setSaving] = React.useState(false)
   190→  const [barcodeFor, setBarcodeFor] = React.useState<number | null>(null)
   191→  const [barcodeVal, setBarcodeVal] = React.useState('')
   192→
   193→  React.useEffect(() => {
   194→    if (orderId == null) { setOrder(null); return }
   195→    let alive = true
   196→    setLoading(true)
   197→    setOrder(null)
   198→    setScan(''); setBarcodeFor(null); setBarcodeVal('')
   199→    api.get<DetailOrder>(`/api/orders/${orderId}`)
   200→      .then((o) => { if (!alive) return; setOrder(o); setRows(initRows(mode, o)) })
   201→      .catch(() => { if (!alive) return; toast.error('خطا در بارگذاری سفارش'); onClose() })
   202→      .finally(() => { if (alive) setLoading(false) })
   203→    return () => { alive = false }
   204→  }, [orderId, mode, onClose])
   205→
   206→  const upd = (id: number, patch: Partial<RowState>) =>
   207→    setRows((r) => (r[id] ? { ...r, [id]: { ...r[id], ...patch } } : r))
   208→
   209→  /* ---- barcode scan: mark item OK with full qty ---- */
   210→  const handleScan = () => {
   211→    const code = scan.trim()
   212→    if (!code || !order) return
   213→    const it = order.items.find((i) => (i.barcode ?? '') === code || (i.product?.barcode ?? '') === code)
   214→    if (!it) { toast.error('قلمی با این بارکد در سفارش یافت نشد'); return }
   215→    const q = mode === 'receive' ? it.qty : (it.deliveredQty ?? it.qty)
   216→    upd(it.id, { status: 'OK', qty: q, note: '' })
   217→    toast.success(`تایید شد ✓ — ${it.name}`)
   218→    setScan('')
   219→  }
   220→
   221→  /* ---- add barcode to an item (product) ---- */
   222→  const saveBarcode = async (itemId: number) => {
   223→    const barcode = barcodeVal.trim()
   224→    if (!barcode || orderId == null) return
   225→    try {
   226→      await api.patch(`/api/orders/${orderId}`, { action: 'addBarcode', userId: user.id, itemId, barcode })
   227→      toast.success('بارکد ذخیره شد ✓')
   228→      setBarcodeVal('')
   229→      setBarcodeFor(null)
   230→      setOrder((o) => (o
   231→        ? { ...o, items: o.items.map((i) => (i.id === itemId && !i.barcode ? { ...i, barcode } : i)) }
   232→        : o))
   233→    } catch (e) {
   234→      toast.error(e instanceof Error ? e.message : 'خطا در ذخیره بارکد')
   235→    }
   236→  }
   237→
   238→  /* ---- recalc footer ---- */
   239→  const totals = React.useMemo(() => {
   240→    if (!order) return { subtotal: 0, vat: 0, total: 0 }
   241→    let subtotal = 0
   242→    for (const it of order.items) {
   243→      const row = rows[it.id]
   244→      if (!row) continue
   245→      const cost = mode === 'receive'
   246→        ? (num(row.printed) ?? it.unitCost)
   247→        : (num(row.final) ?? it.printedPrice ?? it.unitCost)
   248→      subtotal += cost * row.qty
   249→    }
   250→    const vat = subtotal * VAT_RATE
   251→    return { subtotal, vat, total: subtotal + vat }
   252→  }, [order, rows, mode])
   253→
   254→  const delta = order ? totals.total - order.total : 0
   255→  const smallDelta = Math.abs(delta) <= DELTA_TOLERANCE
   256→
   257→  const submit = async () => {
   258→    if (!order) return
   259→    setSaving(true)
   260→    try {
   261→      const items = order.items.map((it) => {
   262→        const row = rows[it.id] ?? { qty: 0, printed: '', final: '', status: 'OK' as ItemStatus, note: '' }
   263→        return mode === 'receive'
   264→          ? {
   265→            id: it.id,
   266→            deliveredQty: row.qty,
   267→            printedPrice: num(row.printed),
   268→            status: row.status,
   269→            note: row.note || null,
   270→            finalCost: num(row.printed),
   271→          }
   272→          : {
   273→            id: it.id,
   274→            confirmedQty: row.qty,
   275→            finalCost: num(row.final),
   276→            status: row.status,
   277→            note: row.note || null,
   278→          }
   279→      })
   280→      await api.patch(`/api/orders/${order.id}`, {
   281→        action: mode === 'receive' ? 'receive' : 'confirm',
   282→        userId: user.id,
   283→        items,
   284→      })
   285→      toast.success(mode === 'receive' ? 'تحویل سفارش ثبت شد ✓' : 'تایید نهایی انبار ثبت شد ✓')
   286→      onDone()
   287→      onClose()
   288→    } catch (e) {
   289→      toast.error(e instanceof Error ? e.message : 'خطا در ثبت')
   290→    } finally {
   291→      setSaving(false)
   292→    }
   293→  }
   294→
   295→  const isToday = order ? order.receivingDate.slice(0, 10) === todayISO() : false
   296→
   297→  return (
   298→    <Modal
   299→      open={orderId != null}
   300→      onClose={onClose}
   301→      wide
   302→      title={mode === 'receive' ? 'دریافت تحویل سفارش' : 'تایید نهایی انبار'}
   303→    >
   304→      {loading || !order ? (
   305→        <Loading label="در حال بارگذاری اقلام سفارش…" />
   306→      ) : (
   307→        <div className="space-y-4">
   308→          {/* ---- header ---- */}
   309→          <div className="flex flex-wrap items-center justify-between gap-2 rounded-2xl border border-[#E4DCC8] bg-white p-4">
   310→            <div>
   311→              <div className="flex items-center gap-2">
   312→                <span className="text-lg font-extrabold text-[#253A2A]">{order.code}</span>
   313→                <StatusBadge status={order.status} />
   314→              </div>
   315→              <div className="mt-0.5 text-sm text-[#6B7A66]">تامین‌کننده: <b className="text-[#3E6B4A]">{order.supplier?.name ?? '—'}</b></div>
   316→            </div>
   317→            <div className="text-left">
   318→              <Badge className={isToday ? 'border-emerald-200 bg-emerald-50 text-emerald-700' : 'border-[#E4DCC8] bg-[#F5F2E8] text-[#4A5A44]'}>
   319→                {isToday ? 'تحویل صبح امروز' : `تحویل: ${fmtJalali(order.receivingDate)}`}
   320→              </Badge>
   321→              <div className="mt-1 text-xs text-[#8A9884]">
   322→                جمع اصلی سفارش: <b className="tabular-nums">{fmtMoney(order.total)}</b> · {toFaDigits(order.items.length)} قلم
   323→              </div>
   324→            </div>
   325→          </div>
   326→
   327→          {/* ---- scan box ---- */}
   328→          <div className="flex items-center gap-2 rounded-xl border border-[#93C572]/50 bg-[#F3F7EF] px-3 py-1.5">
   329→            <ScanLine size={18} className="shrink-0 text-[#3E6B4A]" />
   330→            <input
   331→              autoFocus
   332→              value={scan}
   333→              onChange={(e) => setScan(e.target.value)}
   334→              onKeyDown={(e) => { if (e.key === 'Enter') { e.preventDefault(); handleScan() } }}
   335→              placeholder="بارکد را اسکن کنید و Enter بزنید…"
   336→              dir="ltr"
   337→              className="h-11 w-full bg-transparent text-sm text-[#253A2A] outline-none placeholder:text-[#A8A28C]"
   338→              aria-label="اسکن بارکد"
   339→            />
   340→          </div>
   341→
   342→          {/* ---- item cards ---- */}
   343→          <div className="max-h-[46vh] space-y-3 overflow-y-auto pl-1 pz-scroll">
   344→            {order.items.map((it) => {
   345→              const row = rows[it.id]
   346→              if (!row) return null
   347→              const cost = mode === 'receive'
   348→                ? (num(row.printed) ?? it.unitCost)
   349→                : (num(row.final) ?? it.printedPrice ?? it.unitCost)
   350→              const line = cost * row.qty
   351→              return (
   352→                <Card key={it.id} className="p-3">
   353→                  {/* product head */}
   354→                  <div className="flex items-center gap-3">
   355→                    <ProductImage src={it.product?.imageUrl} name={it.name} size={48} />
   356→                    <div className="min-w-0 flex-1">
   357→                      <div className="truncate text-sm font-bold text-[#253A2A]">{it.name}</div>
   358→                      <div className="font-mono text-[11px] text-[#8A9884]" dir="ltr">{it.barcode || it.product?.barcode || '—'}</div>
   359→                      <div className="text-[11px] text-[#6B7A66]">
   360→                        مقدار سفارش: <b className="tabular-nums">{toFaDigits(it.qty)}</b>
   361→                        {mode === 'receive' && it.deliveredQty != null && <> · دریافتی قبلی: <b className="tabular-nums">{toFaDigits(it.deliveredQty)}</b></>}
   362→                      </div>
   363→                    </div>
   364→                    <div className="shrink-0 text-left">
   365→                      <div className="text-[10px] text-[#8A9884]">جمع خط</div>

   370→                  {/* qty + price */}
   371→                  <div className="mt-3 grid gap-3 sm:grid-cols-2">

   375→                      onChange={(n) => upd(it.id, { qty: n })}

   377→                    {mode === 'receive' ? (
   378→                      <Field label="قیمت چاپ‌شده (تومان)" hint="قیمت چاپ‌شده روی بسته">
   379→                        <input
   380→                          type="number" inputMode="decimal" min={0} step="any"
   381→                          value={row.printed}
   382→                          onChange={(e) => upd(it.id, { printed: e.target.value })}
   383→                          className={inputCls}
   384→                        />
   385→                      </Field>
   386→                    ) : (
   387→                      <Field label="قیمت نهایی واحد (تومان)" hint="پیش‌فرض: قیمت چاپ‌شده / قیمت خرید">
   388→                        <input
   389→                          type="number" inputMode="decimal" min={0} step="any"
   390→                          value={row.final}
   391→                          onChange={(e) => upd(it.id, { final: e.target.value })}
   392→                          className={inputCls}
   393→                        />
   394→                      </Field>
   395→                    )}
   396→                  </div>
   397→
   398→                  {/* status toggles */}
   399→                  <div className="mt-3 grid grid-cols-3 gap-2">
   400→                    {STATUS_OPTS.map((o) => (
   401→                      <button
   402→                        key={o.v} type="button"
   403→                        onClick={() => upd(it.id, { status: o.v })}

   408→                        )}
   409→                      >
   410→                        {o.label}
   411→                      </button>
   412→                    ))}
   413→                  </div>
   414→
   415→                  {/* note when MISSING / REJECTED */}
   416→                  {row.status !== 'OK' && (
   417→                    <div className="mt-3">
   418→                      <Field label="توضیح" hint="دلیل ناقص یا مردود بودن را بنویسید">

   428→
   429→                  {/* add barcode */}
   430→                  <div className="mt-3">
   431→                    {barcodeFor === it.id ? (

   802→/* =============== camera scan modal (BarcodeDetector + graceful fallback) =============== */
   803→
   804→/**
   805→ * Full-screen camera viewfinder for order-barcode scanning.
   806→ * - Feature detection first: no BarcodeDetector / no getUserMedia → «unsupported»
   807→ *   fallback (browser can't do it). getUserMedia failure (denied / no device /
   808→ *   busy / insecure context) → «unavailable» fallback with a specific hint.
   809→ * - Live path: environment-facing stream in a premium viewfinder (gold corner
   810→ *   brackets, pz-cam-beam sweep), code_39 polled ~250 ms; a read is accepted
   811→ *   only when the same raw value appears twice in a row (dedupeRead) →
   812→ *   stop stream → onCode(rawValue) → the ScanBar runs its normal resolve().
   813→ * - All tracks stop on close, on tab-hide (visibilitychange) and on unmount.
   814→ */
   815→function ScanCameraModal({ open, onClose, onCode, onStatusChange }: {
   816→  open: boolean
   817→  onClose: () => void
   818→  onCode: (raw: string) => void
   819→  onStatusChange?: (status: CamStatus) => void
   820→}) {
   821→  const videoRef = React.useRef<HTMLVideoElement | null>(null)
   822→  const streamRef = React.useRef<MediaStream | null>(null)
   823→  const [status, setStatus] = React.useState<CamStatus>('starting')
   824→  const [errHint, setErrHint] = React.useState('')
   825→  const [torchOn, setTorchOn] = React.useState(false) // torch constraint currently applied
   826→  const [torchOk, setTorchOk] = React.useState(false) // device reports torch capability → show toggle
   827→
   828→  const stopStream = React.useCallback(() => {
   829→    const st = streamRef.current
   830→    if (!st) return
   831→    // stopping the tracks also kills an active torch — the explicit state
   832→    // reset just keeps the UI honest for the next open
   833→    setTorchOn(false)
   834→    setTorchOk(false)
   835→    for (const t of st.getTracks()) t.stop()
   836→    streamRef.current = null
   837→  }, [])
   838→
   839→  /** torch toggle — feature-detected upstream; a failed constraint silently
   840→   * keeps the previous state (button simply stays as it was) */
   841→  const toggleTorch = React.useCallback(async () => {
   842→    const track = streamRef.current?.getVideoTracks()[0]
   843→    if (!track) return
   844→    const next = !torchOn
   845→    try {
   846→      await track.applyConstraints({ advanced: [{ torch: next }] } as unknown as MediaTrackConstraints)
   847→      setTorchOn(next)
   848→    } catch {
   849→      /* torch not applied — state stays as it was */
   850→    }
   851→  }, [torchOn])
   852→
   853→  React.useEffect(() => {
   854→    if (!open) return
   855→    let cancelled = false
   856→    let timer: ReturnType<typeof setInterval> | null = null
   857→
   858→    const apply = (s: CamStatus) => {
   859→      setStatus(s)
   860→      onStatusChange?.(s)
   861→    }
   862→
   863→    /** first stable read → stop everything, hand the raw code to the bar */
   864→    const finish = (raw: string) => {
   865→      cancelled = true
   866→      if (timer) { clearInterval(timer); timer = null }
   867→      stopStream()
   868→      onCode(raw)
   869→    }
   870→
   871→    const start = async () => {
   872→      apply('starting')
   873→      setErrHint('')
   874→      const Ctor = window.BarcodeDetector
   875→      if (!Ctor || !navigator.mediaDevices?.getUserMedia) {
   876→        apply('unsupported')
   877→        return
   878→      }
   879→      try {
   880→        const stream = await navigator.mediaDevices.getUserMedia({
   881→          video: { facingMode: 'environment' },
   882→          audio: false,
   883→        })
   884→        if (cancelled) {
   885→          for (const t of stream.getTracks()) t.stop()
   886→          return
   887→        }
   888→        streamRef.current = stream
   889→        const v = videoRef.current
   890→        if (v) {
   891→          v.srcObject = stream
   892→          try { await v.play() } catch { /* muted+playsInline usually autoplay fine */ }
   893→        }
