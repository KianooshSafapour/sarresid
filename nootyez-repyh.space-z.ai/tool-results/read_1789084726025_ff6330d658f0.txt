    57→    57→/** numeric input → number | null (accepts Persian digits too) */
    58→    58→function num(v: string): number | null {
    59→    59→  const t = String(v).trim().replace(/[۰-۹]/g, (d) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d)))
    60→    60→  if (!t) return null
    61→    61→  const n = Number(t)
    62→    62→  return isFinite(n) ? n : null
    63→    63→}

    65→    65→const round1 = (n: number) => Math.round(n * 100) / 100

    67→    67→function initRows(mode: SheetMode, order: DetailOrder): Record<number, RowState> {
    68→    68→  const rec: Record<number, RowState> = {}
    69→    69→  for (const it of order.items) {
    70→    70→    rec[it.id] = {
    71→    71→      qty: mode === 'receive'
    72→    72→        ? (it.deliveredQty ?? it.qty)
    73→    73→        : (it.confirmedQty ?? it.deliveredQty ?? it.qty),
    74→    74→      printed: String(it.printedPrice ?? it.unitCost ?? 0),
    75→    75→      final: String(it.finalCost ?? it.printedPrice ?? it.unitCost ?? 0),
    76→    76→      status: it.status === 'MISSING' || it.status === 'REJECTED' ? it.status : 'OK',
    77→    77→      note: it.note ?? '',
    78→    78→    }
    79→    79→  }
    80→    80→  return rec
    81→    81→}

   123→   123→function DeliverySheet({ orderId, mode, user, onClose, onDone }: {
   124→   124→  orderId: number | null
   125→   125→  mode: SheetMode
   126→   126→  user: PUser
   127→   127→  onClose: () => void
   128→   128→  onDone: () => void
   129→   129→}) {
   130→   130→  const [order, setOrder] = React.useState<DetailOrder | null>(null)
   131→   131→  const [loading, setLoading] = React.useState(false)
   132→   132→  const [rows, setRows] = React.useState<Record<number, RowState>>({})
   133→   133→  const [scan, setScan] = React.useState('')
   134→   134→  const [saving, setSaving] = React.useState(false)
   135→   135→  const [barcodeFor, setBarcodeFor] = React.useState<number | null>(null)
   136→   136→  const [barcodeVal, setBarcodeVal] = React.useState('')
   137→   137→
   138→   138→  React.useEffect(() => {
   139→   139→    if (orderId == null) { setOrder(null); return }
   140→   140→    let alive = true
   141→   141→    setLoading(true)
   142→   142→    setOrder(null)
   143→   143→    setScan(''); setBarcodeFor(null); setBarcodeVal('')
   144→   144→    api.get<DetailOrder>(`/api/orders/${orderId}`)
   145→   145→      .then((o) => { if (!alive) return; setOrder(o); setRows(initRows(mode, o)) })
   146→   146→      .catch(() => { if (!alive) return; toast.error('خطا در بارگذاری سفارش'); onClose() })
   147→   147→      .finally(() => { if (alive) setLoading(false) })
   148→   148→    return () => { alive = false }
   149→   149→  }, [orderId, mode, onClose])
   150→   150→
   151→   151→  const upd = (id: number, patch: Partial<RowState>) =>
   152→   152→    setRows((r) => (r[id] ? { ...r, [id]: { ...r[id], ...patch } } : r))
   153→   153→
   154→   154→  /* ---- barcode scan: mark item OK with full qty ---- */
   155→   155→  const handleScan = () => {
   156→   156→    const code = scan.trim()
   157→   157→    if (!code || !order) return
   158→   158→    const it = order.items.find((i) => (i.barcode ?? '') === code || (i.product?.barcode ?? '') === code)
   159→   159→    if (!it) { toast.error('قلمی با این بارکد در سفارش یافت نشد'); return }
   160→   160→    const q = mode === 'receive' ? it.qty : (it.deliveredQty ?? it.qty)
   161→   161→    upd(it.id, { status: 'OK', qty: q, note: '' })
   162→   162→    toast.success(`تایید شد ✓ — ${it.name}`)
   163→   163→    setScan('')
   164→   164→  }
   165→   165→
   166→   166→  /* ---- add barcode to an item (product) ---- */
   167→   167→  const saveBarcode = async (itemId: number) => {
   168→   168→    const barcode = barcodeVal.trim()
   169→   169→    if (!barcode || orderId == null) return
   170→   170→    try {
   171→   171→      await api.patch(`/api/orders/${orderId}`, { action: 'addBarcode', userId: user.id, itemId, barcode })
   172→   172→      toast.success('بارکد ذخیره شد ✓')
   173→   173→      setBarcodeVal('')
   174→   174→      setBarcodeFor(null)
   175→   175→      setOrder((o) => (o
   176→   176→        ? { ...o, items: o.items.map((i) => (i.id === itemId && !i.barcode ? { ...i, barcode } : i)) }
   177→   177→        : o))
   178→   178→    } catch (e) {
   179→   179→      toast.error(e instanceof Error ? e.message : 'خطا در ذخیره بارکد')
   180→   180→    }
   181→   181→  }
   182→   182→
   183→   183→  /* ---- recalc footer ---- */
   184→   184→  const totals = React.useMemo(() => {
   185→   185→    if (!order) return { subtotal: 0, vat: 0, total: 0 }
   186→   186→    let subtotal = 0
   187→   187→    for (const it of order.items) {
   188→   188→      const row = rows[it.id]
   189→   189→      if (!row) continue
   190→   190→      const cost = mode === 'receive'
   191→   191→        ? (num(row.printed) ?? it.unitCost)
   192→   192→        : (num(row.final) ?? it.printedPrice ?? it.unitCost)
   193→   193→      subtotal += cost * row.qty
   194→   194→    }
   195→   195→    const vat = subtotal * VAT_RATE
   196→   196→    return { subtotal, vat, total: subtotal + vat }
   197→   197→  }, [order, rows, mode])
   198→   198→
   199→   199→  const delta = order ? totals.total - order.total : 0
   200→   200→  const smallDelta = Math.abs(delta) <= DELTA_TOLERANCE
   201→   201→
   202→   202→  const submit = async () => {
   203→   203→    if (!order) return
   204→   204→    setSaving(true)
   205→   205→    try {
   206→   206→      const items = order.items.map((it) => {
   207→   207→        const row = rows[it.id] ?? { qty: 0, printed: '', final: '', status: 'OK' as ItemStatus, note: '' }
   208→   208→        return mode === 'receive'
   209→   209→          ? {
   210→   210→            id: it.id,
   211→   211→            deliveredQty: row.qty,
   212→   212→            printedPrice: num(row.printed),
   213→   213→            status: row.status,
   214→   214→            note: row.note || null,
   215→   215→            finalCost: num(row.printed),
   216→   216→          }
   217→   217→          : {
   218→   218→            id: it.id,
   219→   219→            confirmedQty: row.qty,
   220→   220→            finalCost: num(row.final),
   221→   221→            status: row.status,
   222→   222→            note: row.note || null,
   223→   223→          }
   224→   224→      })
   225→   225→      await api.patch(`/api/orders/${order.id}`, {
   226→   226→        action: mode === 'receive' ? 'receive' : 'confirm',
   227→   227→        userId: user.id,
   228→   228→        items,
   229→   229→      })
   230→   230→      toast.success(mode === 'receive' ? 'تحویل سفارش ثبت شد ✓' : 'تایید نهایی انبار ثبت شد ✓')
   231→   231→      onDone()
   232→   232→      onClose()
   233→   233→    } catch (e) {
   234→   234→      toast.error(e instanceof Error ? e.message : 'خطا در ثبت')
   235→   235→    } finally {
   236→   236→      setSaving(false)
   237→   237→    }
   238→   238→  }
   239→   239→
   240→   240→  const isToday = order ? order.receivingDate.slice(0, 10) === todayISO() : false
   241→   241→
   242→   242→  return (
   243→   243→    <Modal
   244→   244→      open={orderId != null}
   245→   245→      onClose={onClose}
   246→   246→      wide
   247→   247→      title={mode === 'receive' ? 'دریافت تحویل سفارش' : 'تایید نهایی انبار'}
   248→   248→    >
   249→   249→      {loading || !order ? (
   250→   250→        <Loading label="در حال بارگذاری اقلام سفارش…" />
   251→   251→      ) : (
   252→   252→        <div className="space-y-4">
   253→   253→          {/* ---- header ---- */}
   254→   254→          <div className="flex flex-wrap items-center justify-between gap-2 rounded-2xl border border-[#E4DCC8] bg-white p-4">
   255→   255→            <div>
   256→   256→              <div className="flex items-center gap-2">
   257→   257→                <span className="text-lg font-extrabold text-[#253A2A]">{order.code}</span>
   258→   258→                <StatusBadge status={order.status} />
   259→   259→              </div>
   260→   260→              <div className="mt-0.5 text-sm text-[#6B7A66]">تامین‌کننده: <b className="text-[#3E6B4A]">{order.supplier?.name ?? '—'}</b></div>
   261→   261→            </div>
   262→   262→            <div className="text-left">
   263→   263→              <Badge className={isToday ? 'border-emerald-200 bg-emerald-50 text-emerald-700' : 'border-[#E4DCC8] bg-[#F5F2E8] text-[#4A5A44]'}>
   264→   264→                {isToday ? 'تحویل صبح امروز' : `تحویل: ${fmtJalali(order.receivingDate)}`}
   265→   265→              </Badge>
   266→   266→              <div className="mt-1 text-xs text-[#8A9884]">
   267→   267→                جمع اصلی سفارش: <b className="tabular-nums">{fmtMoney(order.total)}</b> · {toFaDigits(order.items.length)} قلم
   268→   268→              </div>
   269→   269→            </div>
   270→   270→          </div>
   271→   271→
   272→   272→          {/* ---- scan box ---- */}
   273→   273→          <div className="flex items-center gap-2 rounded-xl border border-[#93C572]/50 bg-[#F3F7EF] px-3 py-1.5">
   274→   274→            <ScanLine size={18} className="shrink-0 text-[#3E6B4A]" />
   275→   275→            <input
   276→   276→              autoFocus
   277→   277→              value={scan}
   278→   278→              onChange={(e) => setScan(e.target.value)}
   279→   279→              onKeyDown={(e) => { if (e.key === 'Enter') { e.preventDefault(); handleScan() } }}
   280→   280→              placeholder="بارکد را اسکن کنید و Enter بزنید…"
   281→   281→              dir="ltr"
   282→   282→              className="h-11 w-full bg-transparent text-sm text-[#253A2A] outline-none placeholder:text-[#A8A28C]"
   283→   283→              aria-label="اسکن بارکد"
   284→   284→            />
   285→   285→          </div>
   286→   286→
   287→   287→          {/* ---- item cards ---- */}
   288→   288→          <div className="max-h-[46vh] space-y-3 overflow-y-auto pl-1 pz-scroll">
   289→   289→            {order.items.map((it) => {
   290→   290→              const row = rows[it.id]
   291→   291→              if (!row) return null
   292→   292→              const cost = mode === 'receive'
   293→   293→                ? (num(row.printed) ?? it.unitCost)
   294→   294→                : (num(row.final) ?? it.printedPrice ?? it.unitCost)
   295→   295→              const line = cost * row.qty
   296→   296→              return (
   297→   297→                <Card key={it.id} className="p-3">
   298→   298→                  {/* product head */}
   299→   299→                  <div className="flex items-center gap-3">
   300→   300→                    <ProductImage src={it.product?.imageUrl} name={it.name} size={48} />
   301→   301→                    <div className="min-w-0 flex-1">
   302→   302→                      <div className="truncate text-sm font-bold text-[#253A2A]">{it.name}</div>
   303→   303→                      <div className="font-mono text-[11px] text-[#8A9884]" dir="ltr">{it.barcode || it.product?.barcode || '—'}</div>
   304→   304→                      <div className="text-[11px] text-[#6B7A66]">
   305→   305→                        مقدار سفارش: <b className="tabular-nums">{toFaDigits(it.qty)}</b>
   306→   306→                        {mode === 'receive' && it.deliveredQty != null && <> · دریافتی قبلی: <b className="tabular-nums">{toFaDigits(it.deliveredQty)}</b></>}
   307→   307→                      </div>
   308→   308→                    </div>
   309→   309→                    <div className="shrink-0 text-left">
   310→   310→                      <div className="text-[10px] text-[#8A9884]">جمع خط</div>
   311→   311→                      <div className="text-xs font-bold tabular-nums text-[#3E6B4A]">{fmtMoney(line)}</div>
   312→   312→                    </div>
   313→   313→                  </div>
   314→   314→
   315→   315→                  {/* qty + price */}
   316→   316→                  <div className="mt-3 grid gap-3 sm:grid-cols-2">
   317→   317→                    <QtyStepper
   318→   318→                      label={mode === 'receive' ? 'تعداد دریافتی' : 'تعداد تایید نهایی'}
   319→   319→                      value={row.qty}
   320→   320→                      onChange={(n) => upd(it.id, { qty: n })}
   321→   321→                    />
   322→   322→                    {mode === 'receive' ? (
   323→   323→                      <Field label="قیمت چاپ‌شده (تومان)" hint="قیمت چاپ‌شده روی بسته">
   324→   324→                        <input
   325→   325→                          type="number" inputMode="decimal" min={0} step="any"
   326→   326→                          value={row.printed}
   327→   327→                          onChange={(e) => upd(it.id, { printed: e.target.value })}
   328→   328→                          className={inputCls}
   329→   329→                        />
   330→   330→                      </Field>
   331→   331→                    ) : (
   332→   332→                      <Field label="قیمت نهایی واحد (تومان)" hint="پیش‌فرض: قیمت چاپ‌شده / قیمت خرید">
   333→   333→                        <input
   334→   334→                          type="number" inputMode="decimal" min={0} step="any"
   335→   335→                          value={row.final}
   336→   336→                          onChange={(e) => upd(it.id, { final: e.target.value })}
   337→   337→                          className={inputCls}
   338→   338→                        />
   339→   339→                      </Field>
   340→   340→                    )}
   341→   341→                  </div>
   342→   342→
   343→   343→                  {/* status toggles */}
   344→   344→                  <div className="mt-3 grid grid-cols-3 gap-2">
   345→   345→                    {STATUS_OPTS.map((o) => (
   346→   346→                      <button
   347→   347→                        key={o.v} type="button"
   348→   348→                        onClick={() => upd(it.id, { status: o.v })}
   349→   349→                        aria-pressed={row.status === o.v}
   350→   350→                        className={cn(
   351→   351→                          'min-h-[44px] rounded-xl border-2 px-2 text-sm font-bold transition-all active:scale-[0.98]',
   352→   352→                          row.status === o.v ? o.activeCls : o.idleCls
   353→   353→                        )}
   354→   354→                      >
   355→   355→                        {o.label}
   356→   356→                      </button>
   357→   357→                    ))}
   358→   358→                  </div>
   359→   359→
   360→   360→                  {/* note when MISSING / REJECTED */}
   361→   361→                  {row.status !== 'OK' && (
   362→   362→                    <div className="mt-3">
   363→   363→                      <Field label="توضیح" hint="دلیل ناقص یا مردود بودن را بنویسید">
   364→   364→                        <input
   365→   365→                          value={row.note}
   366→   366→                          onChange={(e) => upd(it.id, { note: e.target.value })}
   367→   367→                          placeholder="مثلاً: ۲ عدد شکسته بود"
   368→   368→                          className={inputCls}
   369→   369→                        />
   370→   370→                      </Field>
   371→   371→                    </div>
   372→   372→                  )}
   373→   373→
   374→   374→                  {/* add barcode */}
   375→   375→                  <div className="mt-3">
   376→   376→                    {barcodeFor === it.id ? (
   377→   377→                      <div className="flex flex-col gap-2 sm:flex-row">
   378→   378→                        <input
   379→   379→                          autoFocus
   380→   380→                          value={barcodeVal}
   381→   381→                          onChange={(e) => setBarcodeVal(e.target.value)}
   382→   382→                          onKeyDown={(e) => { if (e.key === 'Enter') { e.preventDefault(); saveBarcode(it.id) } }}
   383→   383→                          placeholder="بارکد جدید را اسکن/وارد کنید…"
   384→   384→                          dir="ltr"
   385→   385→                          className={cn(inputCls, 'font-mono')}
   386→   386→                        />
