    48→/** downscale an image file via canvas → JPEG dataUrl (max 320px, quality 0.8) */
    49→async function fileToDataUrl(file: File): Promise<string> {
    50→  const dataUrl = await new Promise<string>((resolve, reject) => {
    51→    const r = new FileReader()
    52→    r.onload = () => resolve(String(r.result))
    53→    r.onerror = () => reject(new Error('read failed'))
    54→    r.readAsDataURL(file)
    55→  })
    56→  const img = document.createElement('img')
    57→  await new Promise<void>((resolve, reject) => {
    58→    img.onload = () => resolve()
    59→    img.onerror = () => reject(new Error('decode failed'))
    60→    img.src = dataUrl
    61→  })
    62→  const max = 320
    63→  const scale = Math.min(1, max / Math.max(img.width || 1, img.height || 1))
    64→  const w = Math.max(1, Math.round(img.width * scale))
    65→  const h = Math.max(1, Math.round(img.height * scale))
    66→  const canvas = document.createElement('canvas')
    67→  canvas.width = w
    68→  canvas.height = h
    69→  canvas.getContext('2d')?.drawImage(img, 0, 0, w, h)
    70→  return canvas.toDataURL('image/jpeg', 0.8)
    71→}
