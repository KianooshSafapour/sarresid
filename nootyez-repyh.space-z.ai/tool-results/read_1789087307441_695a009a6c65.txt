    58→/**
    59→ * Normalize a scanned/typed order code for matching:
    60→ * strip Code39 start/stop `*` and all whitespace, map Persian/Arabic digits
    61→ * to ASCII (manual entry), uppercase. (Inline equivalent of sanitizeCode39
    62→ * from lib/code39 — that one keeps spaces, which scanners do send, and its
    63→ * charset filter would mangle the code shown in error messages.)
    64→ */
    65→function normalizeOrderCode(raw: string): string {
    66→  return String(raw ?? '')
    67→    .replace(/[\s*]+/g, '')
    68→    .replace(/[۰-۹]/g, (d) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d)))
    69→    .replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
    70→    .toUpperCase()
    71→}
    72→
    73→/** numeric input → number | null (accepts Persian digits too) */
    74→function num(v: string): number | null {
    75→  const t = String(v).trim().replace(/[۰-۹]/g, (d) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d)))
    76→  if (!t) return null
    77→  const n = Number(t)
    78→  return isFinite(n) ? n : null
    79→}
    80→
    81→const round1 = (n: number) => Math.round(n * 100) / 100
    82→
    83→function initRows(mode: SheetMode, order: DetailOrder): Record<number, RowState> {
    84→  const rec: Record<number, RowState> = {}
    85→  for (const it of order.items) {
    86→    rec[it.id] = {
    87→      qty: mode === 'receive'
    88→        ? (it.deliveredQty ?? it.qty)
    89→        : (it.confirmedQty ?? it.deliveredQty ?? it.qty),
    90→      printed: String(it.printedPrice ?? it.unitCost ?? 0),
    91→      final: String(it.finalCost ?? it.printedPrice ?? it.unitCost ?? 0),
    92→      status: it.status === 'MISSING' || it.status === 'REJECTED' ? it.status : 'OK',
    93→      note: it.note ?? '',
    94→    }
    95→  }
    96→  return rec
    97→}

   111→      <div className="flex items-center gap-1.5">
   112→        <button
   113→          type="button" aria-label="کاهش تعداد"
   114→          onClick={() => onChange(Math.max(0, round1(value - 1)))}
   115→          className="flex h-11 w-11 shrink-0 items-center justify-center rounded-xl border border-[#D8D2BC] bg-white text-[#4A5A44] transition hover:bg-[#F3F7EF] active:scale-95"
   116→        >
   117→          <Minus size={16} />
   118→        </button>
   119→        <input
   120→          type="number" inputMode="decimal" min={0} step="any" aria-label={label}
   121→          value={String(value)}
   122→          onChange={(e) => { const n = Number(e.target.value); onChange(isFinite(n) && n >= 0 ? round1(n) : 0) }}
   123→          className="h-11 w-full min-w-0 rounded-xl border border-[#D8D2BC] bg-white text-center text-sm font-bold tabular-nums text-[#253A2A] outline-none focus:border-[#5F8F55] focus:ring-2 focus:ring-[#93C572]/30"
   124→        />
   125→        <button
   126→          type="button" aria-label="افزایش تعداد"
   127→          onClick={() => onChange(round1(value + 1))}
   128→          className="flex h-11 w-11 shrink-0 items-center justify-center rounded-xl border border-[#D8D2BC] bg-white text-[#4A5A44] transition hover:bg-[#F3F7EF] active:scale-95"
   129→        >
   130→          <Plus size={16} />
   131→        </button>
   132→      </div>
   133→    </div>
   134→  )
   135→}

   138→
   139→function DeliverySheet({ orderId, mode, user, onClose, onDone }: {
   140→  orderId: number | null
   141→  mode: SheetMode
   142→  user: PUser
   143→  onClose: () => void
   144→  onDone: () => void
   145→}) {
   146→  const [order, setOrder] = React.useState<DetailOrder | null>(null)
   147→  const [loading, setLoading] = React.useState(false)
   148→  const [rows, setRows] = React.useState<Record<number, RowState>>({})
   149→  const [scan, setScan] = React.useState('')
   150→  const [saving, setSaving] = React.useState(false)
   151→  const [barcodeFor, setBarcodeFor] = React.useState<number | null>(null)
   152→  const [barcodeVal, setBarcodeVal] = React.useState('')
   153→
   154→  React.useEffect(() => {
   155→    if (orderId == null) { setOrder(null); return }
   156→    let alive = true
   157→    setLoading(true)
   158→    setOrder(null)
   159→    setScan(''); setBarcodeFor(null); setBarcodeVal('')
   160→    api.get<DetailOrder>(`/api/orders/${orderId}`)
   161→      .then((o) => { if (!alive) return; setOrder(o); setRows(initRows(mode, o)) })
   162→      .catch(() => { if (!alive) return; toast.error('خطا در بارگذاری سفارش'); onClose() })
   163→      .finally(() => { if (alive) setLoading(false) })
   164→    return () => { alive = false }
   165→  }, [orderId, mode, onClose])
   166→
   167→  const upd = (id: number, patch: Partial<RowState>) =>
   168→    setRows((r) => (r[id] ? { ...r, [id]: { ...r[id], ...patch } } : r))
   169→
   170→  /* ---- barcode scan: mark item OK with full qty ---- */
   171→  const handleScan = () => {
   172→    const code = scan.trim()
   173→    if (!code || !order) return
   174→    const it = order.items.find((i) => (i.barcode ?? '') === code || (i.product?.barcode ?? '') === code)
   175→    if (!it) { toast.error('قلمی با این بارکد در سفارش یافت نشد'); return }
   176→    const q = mode === 'receive' ? it.qty : (it.deliveredQty ?? it.qty)
   177→    upd(it.id, { status: 'OK', qty: q, note: '' })
   178→    toast.success(`تایید شد ✓ — ${it.name}`)
   179→    setScan('')
   180→  }
   181→
   182→  /* ---- add barcode to an item (product) ---- */
   183→  const saveBarcode = async (itemId: number) => {
   184→    const barcode = barcodeVal.trim()
   185→    if (!barcode || orderId == null) return
   186→    try {
   187→      await api.patch(`/api/orders/${orderId}`, { action: 'addBarcode', userId: user.id, itemId, barcode })
   188→      toast.success('بارکد ذخیره شد ✓')
   189→      setBarcodeVal('')
   190→      setBarcodeFor(null)
   191→      setOrder((o) => (o
   192→        ? { ...o, items: o.items.map((i) => (i.id === itemId && !i.barcode ? { ...i, barcode } : i)) }
   193→        : o))
   194→    } catch (e) {
   195→      toast.error(e instanceof Error ? e.message : 'خطا در ذخیره بارکد')
   196→    }
   197→  }
   198→
   199→  /* ---- recalc footer ---- */
   200→  const totals = React.useMemo(() => {
   201→    if (!order) return { subtotal: 0, vat: 0, total: 0 }
   202→    let subtotal = 0
   203→    for (const it of order.items) {
   204→      const row = rows[it.id]
   205→      if (!row) continue
   206→      const cost = mode === 'receive'
   207→        ? (num(row.printed) ?? it.unitCost)
   208→        : (num(row.final) ?? it.printedPrice ?? it.unitCost)
   209→      subtotal += cost * row.qty
   210→    }
   211→    const vat = subtotal * VAT_RATE
   212→    return { subtotal, vat, total: subtotal + vat }
   213→  }, [order, rows, mode])
   214→
   215→  const delta = order ? totals.total - order.total : 0
   216→  const smallDelta = Math.abs(delta) <= DELTA_TOLERANCE
   217→
   218→  const submit = async () => {
   219→    if (!order) return
   220→    setSaving(true)
   221→    try {
   222→      const items = order.items.map((it) => {
   223→        const row = rows[it.id] ?? { qty: 0, printed: '', final: '', status: 'OK' as ItemStatus, note: '' }
   224→        return mode === 'receive'
   225→          ? {
   226→            id: it.id,
   227→            deliveredQty: row.qty,
   228→            printedPrice: num(row.printed),
   229→            status: row.status,
   230→            note: row.note || null,
   231→            finalCost: num(row.printed),
   232→          }
   233→          : {
   234→            id: it.id,
   235→            confirmedQty: row.qty,
   236→            finalCost: num(row.final),
   237→            status: row.status,
   238→            note: row.note || null,
   239→          }
   240→      })
   241→      await api.patch(`/api/orders/${order.id}`, {
   242→        action: mode === 'receive' ? 'receive' : 'confirm',
   243→        userId: user.id,
   244→        items,
   245→      })
   246→      toast.success(mode === 'receive' ? 'تحویل سفارش ثبت شد ✓' : 'تایید نهایی انبار ثبت شد ✓')
   247→      onDone()
   248→      onClose()
   249→    } catch (e) {
   250→      toast.error(e instanceof Error ? e.message : 'خطا در ثبت')
   251→    } finally {
   252→      setSaving(false)
   253→    }
   254→  }
   255→
   256→  const isToday = order ? order.receivingDate.slice(0, 10) === todayISO() : false
   257→
   258→  return (
   259→    <Modal
   260→      open={orderId != null}
   261→      onClose={onClose}
   262→      wide
   263→      title={mode === 'receive' ? 'دریافت تحویل سفارش' : 'تایید نهایی انبار'}
   264→    >
   265→      {loading || !order ? (
   266→        <Loading label="در حال بارگذاری اقلام سفارش…" />
   267→      ) : (
   268→        <div className="space-y-4">
   269→          {/* ---- header ---- */}
   270→          <div className="flex flex-wrap items-center justify-between gap-2 rounded-2xl border border-[#E4DCC8] bg-white p-4">
   271→            <div>
   272→              <div className="flex items-center gap-2">
   273→                <span className="text-lg font-extrabold text-[#253A2A]">{order.code}</span>
   274→                <StatusBadge status={order.status} />
   275→              </div>
   276→              <div className="mt-0.5 text-sm text-[#6B7A66]">تامین‌کننده: <b className="text-[#3E6B4A]">{order.supplier?.name ?? '—'}</b></div>
   277→            </div>
   278→            <div className="text-left">
   279→              <Badge className={isToday ? 'border-emerald-200 bg-emerald-50 text-emerald-700' : 'border-[#E4DCC8] bg-[#F5F2E8] text-[#4A5A44]'}>
   280→                {isToday ? 'تحویل صبح امروز' : `تحویل: ${fmtJalali(order.receivingDate)}`}
   281→              </Badge>
   282→              <div className="mt-1 text-xs text-[#8A9884]">
   283→                جمع اصلی سفارش: <b className="tabular-nums">{fmtMoney(order.total)}</b> · {toFaDigits(order.items.length)} قلم
   284→              </div>
   285→            </div>
   286→          </div>
   287→
   288→          {/* ---- scan box ---- */}
   289→          <div className="flex items-center gap-2 rounded-xl border border-[#93C572]/50 bg-[#F3F7EF] px-3 py-1.5">
   290→            <ScanLine size={18} className="shrink-0 text-[#3E6B4A]" />
   291→            <input
   292→              autoFocus
   293→              value={scan}
   294→              onChange={(e) => setScan(e.target.value)}
   295→              onKeyDown={(e) => { if (e.key === 'Enter') { e.preventDefault(); handleScan() } }}
   296→              placeholder="بارکد را اسکن کنید و Enter بزنید…"
   297→              dir="ltr"
   298→              className="h-11 w-full bg-transparent text-sm text-[#253A2A] outline-none placeholder:text-[#A8A28C]"
   299→              aria-label="اسکن بارکد"
   300→            />
   301→          </div>
   302→
   303→          {/* ---- item cards ---- */}
   304→          <div className="max-h-[46vh] space-y-3 overflow-y-auto pl-1 pz-scroll">
   305→            {order.items.map((it) => {
   306→              const row = rows[it.id]
   307→              if (!row) return null
   308→              const cost = mode === 'receive'
   309→                ? (num(row.printed) ?? it.unitCost)
   310→                : (num(row.final) ?? it.printedPrice ?? it.unitCost)
   311→              const line = cost * row.qty
   312→              return (
   313→                <Card key={it.id} className="p-3">
   314→                  {/* product head */}
   315→                  <div className="flex items-center gap-3">
   316→                    <ProductImage src={it.product?.imageUrl} name={it.name} size={48} />
   317→                    <div className="min-w-0 flex-1">
   318→                      <div className="truncate text-sm font-bold text-[#253A2A]">{it.name}</div>
   319→                      <div className="font-mono text-[11px] text-[#8A9884]" dir="ltr">{it.barcode || it.product?.barcode || '—'}</div>
   320→                      <div className="text-[11px] text-[#6B7A66]">
   321→                        مقدار سفارش: <b className="tabular-nums">{toFaDigits(it.qty)}</b>
   322→                        {mode === 'receive' && it.deliveredQty != null && <> · دریافتی قبلی: <b className="tabular-nums">{toFaDigits(it.deliveredQty)}</b></>}
   323→                      </div>
   324→                    </div>
   325→                    <div className="shrink-0 text-left">
   326→                      <div className="text-[10px] text-[#8A9884]">جمع خط</div>
   327→                      <div className="text-xs font-bold tabular-nums text-[#3E6B4A]">{fmtMoney(line)}</div>
   328→                    </div>
   329→                  </div>
   330→
   331→                  {/* qty + price */}
   332→                  <div className="mt-3 grid gap-3 sm:grid-cols-2">
   333→                    <QtyStepper
   334→                      label={mode === 'receive' ? 'تعداد دریافتی' : 'تعداد تایید نهایی'}
   335→                      value={row.qty}
   336→                      onChange={(n) => upd(it.id, { qty: n })}
   337→                    />
   338→                    {mode === 'receive' ? (
   339→                      <Field label="قیمت چاپ‌شده (تومان)" hint="قیمت چاپ‌شده روی بسته">
   340→                        <input
   341→                          type="number" inputMode="decimal" min={0} step="any"
   342→                          value={row.printed}
   343→                          onChange={(e) => upd(it.id, { printed: e.target.value })}
   344→                          className={inputCls}
   345→                        />
   346→                      </Field>
   347→                    ) : (
   348→                      <Field label="قیمت نهایی واحد (تومان)" hint="پیش‌فرض: قیمت چاپ‌شده / قیمت خرید">
   349→                        <input
   350→                          type="number" inputMode="decimal" min={0} step="any"
   351→                          value={row.final}
   352→                          onChange={(e) => upd(it.id, { final: e.target.value })}
   353→                          className={inputCls}
   354→                        />
   355→                      </Field>
   356→                    )}
   357→                  </div>
   358→
   359→                  {/* status toggles */}
   360→                  <div className="mt-3 grid grid-cols-3 gap-2">
   361→                    {STATUS_OPTS.map((o) => (
   362→                      <button
   363→                        key={o.v} type="button"
   364→                        onClick={() => upd(it.id, { status: o.v })}
   365→                        aria-pressed={row.status === o.v}
   366→                        className={cn(
   367→                          'min-h-[44px] rounded-xl border-2 px-2 text-sm font-bold transition-all active:scale-[0.98]',
   368→                          row.status === o.v ? o.activeCls : o.idleCls
   369→                        )}
   370→                      >
   371→                        {o.label}
   372→                      </button>
   373→                    ))}
   374→                  </div>
   375→
   376→                  {/* note when MISSING / REJECTED */}
   377→                  {row.status !== 'OK' && (
   378→                    <div className="mt-3">
   379→                      <Field label="توضیح" hint="دلیل ناقص یا مردود بودن را بنویسید">
   380→                        <input
   381→                          value={row.note}
   382→                          onChange={(e) => upd(it.id, { note: e.target.value })}
   383→                          placeholder="مثلاً: ۲ عدد شکسته بود"
   384→                          className={inputCls}
   385→                        />
   386→                      </Field>
   387→                    </div>
   388→                  )}
   389→
   390→                  {/* add barcode */}
   391→                  <div className="mt-3">
   392→                    {barcodeFor === it.id ? (
   393→                      <div className="flex flex-col gap-2 sm:flex-row">
   394→                        <input
   395→                          autoFocus
   396→                          value={barcodeVal}
   397→                          onChange={(e) => setBarcodeVal(e.target.value)}
   398→                          onKeyDown={(e) => { if (e.key === 'Enter') { e.preventDefault(); saveBarcode(it.id) } }}
   399→                          placeholder="بارکد جدید را اسکن/وارد کنید…"
   400→                          dir="ltr"
   401→                          className={cn(inputCls, 'font-mono')}
   402→                        />
   403→                        <div className="flex gap-2">
   404→                          <PrimaryButton type="button" className="min-h-[44px] flex-1" onClick={() => saveBarcode(it.id)}>ذخیره بارکد</PrimaryButton>
   405→                          <GhostButton type="button" className="min-h-[44px]" onClick={() => { setBarcodeFor(null); setBarcodeVal('') }}>لغو</GhostButton>
   406→                        </div>
   407→                      </div>
   408→                    ) : (
   409→                      <GhostButton type="button" className="min-h-[44px] w-full sm:w-auto" onClick={() => { setBarcodeFor(it.id); setBarcodeVal('') }}>
   410→                        افزودن بارکد
   411→                      </GhostButton>
   412→                    )}
   413→                  </div>
   414→                </Card>
   415→              )
   416→            })}
   417→          </div>
   418→
   419→          {/* ---- recalc footer ---- */}
   420→          <Card className="border-[#EAD9A8] bg-gradient-to-br from-[#FFFDF5] to-[#FBF3DC] p-4">
   421→            <div className="grid grid-cols-3 gap-2 text-center">
   422→              <div>
   423→                <div className="text-[11px] text-[#8A6508]">جمع اقلام</div>
   424→                <div className="text-sm font-bold tabular-nums text-[#253A2A]">{fmtMoney(totals.subtotal)}</div>
   425→              </div>
   426→              <div>
   427→                <div className="text-[11px] text-[#8A6508]">مالیات بر ارزش افزوده (۹٪)</div>
   428→                <div className="text-sm font-bold tabular-nums text-[#253A2A]">{fmtMoney(totals.vat)}</div>

   436→              <span
   437→                className={cn(
   438→                  'inline-flex items-center gap-1 rounded-full border px-3 py-1.5 text-xs font-bold',
   439→                  smallDelta ? 'border-emerald-200 bg-emerald-50 text-emerald-700' : 'border-amber-300 bg-amber-50 text-amber-800'
   440→                )}

   464→type ScanOutcome = { kind: 'ok' | 'warn' | 'err'; text: string }
   465→
   466→const SCAN_RECENT_KEY = 'hz_scan_recent'
   467→
   468→function readRecentCodes(): string[] {
   469→  try {
   470→    const raw = window.sessionStorage.getItem(SCAN_RECENT_KEY)
   471→    const arr = raw ? JSON.parse(raw) : []
   472→    return Array.isArray(arr) ? arr.filter((c) => typeof c === 'string').slice(0, 3) : []
   473→  } catch {
   474→    return []
   475→  }
   476→}
   477→
   478→function writeRecentCodes(codes: string[]) {
   479→  try {
   480→    window.sessionStorage.setItem(SCAN_RECENT_KEY, JSON.stringify(codes.slice(0, 3)))
   481→  } catch {
   482→    /* private mode etc. */
   483→  }
   484→}

   486→/**
   487→ * Gold/olive scan bar at the top of Deliveries. Barcode scanners behave like
   488→ * keyboards and send Enter after the code, so the input submits on Enter;
   489→ * a gold button covers manual entry. Outcomes come from the parent's
   490→ * `resolve` (match → open detail, order-without-delivery → hint, unknown →
   491→ * error + shake). Keeps the last 3 resolved codes as re-open chips (persisted
   492→ * in sessionStorage so the trail survives the detail open/close round-trip).
   493→ */
   494→function ScanBar({ resolve }: { resolve: (rawCode: string) => Promise<ScanOutcome> }) {
   495→  const [value, setValue] = React.useState('')
   496→  const [outcome, setOutcome] = React.useState<ScanOutcome | null>(null)
   497→  const [looking, setLooking] = React.useState(false)
   498→  const [shaking, setShaking] = React.useState(false)
   499→  const [recent, setRecent] = React.useState<string[]>([])   // hydrated after mount (SSR-safe)
   500→  const recentRef = React.useRef<string[]>([])               // mirror — survives unmount-timing races
   501→  const inputRef = React.useRef<HTMLInputElement>(null)
   502→  const hideTimer = React.useRef<ReturnType<typeof setTimeout> | null>(null)
   503→  const shakeTimer = React.useRef<ReturnType<typeof setTimeout> | null>(null)
   504→
   505→  React.useEffect(() => {
   506→    const saved = readRecentCodes()
   507→    recentRef.current = saved
   508→    setRecent(saved)
   509→    return () => {
   510→      if (hideTimer.current) clearTimeout(hideTimer.current)
   511→      if (shakeTimer.current) clearTimeout(shakeTimer.current)
   512→    }
   513→  }, [])
   514→
   515→  /** record a resolved code — writes storage imperatively first, because a
   516→   * successful resolve may unmount this bar in the same commit (detail opens)
   517→   * and React discards pending updates (never running updater side-effects). */
   518→  const addRecent = (code: string) => {
   519→    const next = [code, ...recentRef.current.filter((c) => c !== code)].slice(0, 3)
   520→    recentRef.current = next
   521→    writeRecentCodes(next)
   522→    setRecent(next)
   523→  }
   524→
   525→  const submit = async (rawOverride?: string) => {
   526→    if (looking) return
   527→    const code = normalizeOrderCode(rawOverride ?? value)
   528→    if (!code) return // empty input → ignore silently
   529→    if (hideTimer.current) { clearTimeout(hideTimer.current); hideTimer.current = null }
   530→    setOutcome(null)
   531→    setLooking(true)
   532→    try {
   533→      const res = await resolve(code)
   534→      setOutcome(res)
   535→      if (res.kind === 'err') {
   536→        // miss → shake + keep the input focused for the next scan
   537→        setShaking(false)
   538→        if (shakeTimer.current) clearTimeout(shakeTimer.current)
   539→        shakeTimer.current = setTimeout(() => setShaking(false), 520)
   540→        requestAnimationFrame(() => setShaking(true))
   541→        inputRef.current?.focus()
   542→      } else {
   543→        addRecent(code)
   544→        if (res.kind === 'warn') inputRef.current?.focus()
   545→        if (res.kind === 'ok') {
   546→          hideTimer.current = setTimeout(() => setOutcome(null), 6000)
   547→        }
   548→      }
   549→    } finally {
   550→      setLooking(false)
   551→      setValue('')
   552→    }
   553→  }
   554→
   555→  return (
   556→    <section
   557→      aria-label="اسکن بارکد تحویل"
   558→      className={cn(
   559→        'relative mb-4 overflow-hidden rounded-2xl border border-[#EAD9A8] bg-gradient-to-l from-[#F3F7EF] via-[#FBF9F3] to-[#FFFDF5] p-4 shadow-[0_2px_14px_-4px_rgba(90,74,32,0.14)]',
   560→        shaking && 'pz-shake'
   561→      )}
   562→    >
   563→      {/* header: title + hint + example legend */}
   564→      <div className="mb-3 flex flex-wrap items-center justify-between gap-2">
   565→        <div className="flex items-center gap-2.5">
   566→          <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-xl bg-gradient-to-br from-[#B8860B] to-[#DAA520] text-white shadow-md shadow-amber-700/20">
   567→            <ScanLine size={20} />
   568→          </span>
   569→          <div>
   570→            <div className="text-sm font-extrabold text-[#253A2A]">اسکن تحویل | Scan-in</div>
   571→            <div className="text-[11px] text-[#8A9884]">بارکد حک‌شده روی چاپ سفارش را اسکن کنید</div>
   572→          </div>
   573→        </div>
   574→        <Badge className="pz-barcode border-[#EAD9A8] bg-[#FFFDF5] text-[11px] text-[#8A6508]">مثال: HZ-1005</Badge>
   575→      </div>
   576→
   577→      {/* input row: scanner line + icon + LTR mono input + submit button */}
   578→      <div className="flex flex-col gap-2 sm:flex-row">
   579→        <div
   580→          className={cn(
   581→            'relative flex-1 rounded-xl border border-[#E4DCC8] bg-white transition-all',
   582→            'focus-within:border-[#DAA520] focus-within:shadow-[0_0_0_4px_rgba(218,165,32,0.16)]'
   583→          )}
   584→        >
   585→          <span aria-hidden className="pz-scanline left-1.5" />
   586→          <ScanLine size={18} className="pointer-events-none absolute left-5 top-1/2 -translate-y-1/2 text-[#B8860B]/70" aria-hidden />
   587→          <input
   588→            ref={inputRef}
   589→            value={value}
   590→            onChange={(e) => setValue(e.target.value)}
   591→            onKeyDown={(e) => { if (e.key === 'Enter') { e.preventDefault(); void submit() } }}
   592→            placeholder="کد سفارش را اسکن یا وارد کنید… | Scan order code"
   593→            dir="ltr"
   594→            aria-label="کد سفارش را اسکن یا وارد کنید"
   595→            autoComplete="off"
   596→            autoCapitalize="characters"
   597→            spellCheck={false}
   598→            className="h-12 min-h-[48px] w-full bg-transparent pl-12 pr-4 pz-barcode text-base font-bold tracking-widest text-[#253A2A] outline-none placeholder:font-sans placeholder:text-sm placeholder:font-normal placeholder:tracking-normal placeholder:text-[#A8A28C]"
   599→          />
   600→        </div>
   601→        <GoldButton
   602→          type="button"
   603→          onClick={() => void submit()}
   604→          disabled={looking}
   605→          className="min-h-[48px] shrink-0 px-5"
   606→        >
   607→          {looking ? <Spinner /> : <ScanLine size={17} />} باز کردن
   608→        </GoldButton>
   609→      </div>
   610→
   611→      {/* outcome banners */}
   612→      {outcome?.kind === 'ok' && (
   613→        <div role="status" className="mt-2.5 flex items-center gap-2 rounded-xl border border-[#EAD9A8] bg-[#FFFDF5] px-3 py-2.5 text-sm font-bold text-[#8A6508]">
   614→          <CheckCircle2 size={17} className="shrink-0" />
   615→          <span>{outcome.text}</span>
   616→        </div>
   617→      )}
   618→      {outcome?.kind === 'warn' && (
   619→        <div role="status" className="mt-2.5 flex items-center gap-2 rounded-xl border border-amber-300 bg-amber-50 px-3 py-2.5 text-sm font-bold text-amber-800">
   620→          <AlertTriangle size={17} className="shrink-0" />
   621→          <span>{outcome.text}</span>
   622→        </div>
   623→      )}
   624→      {outcome?.kind === 'err' && (
   625→        <div role="alert" className="mt-2.5 flex items-center gap-2 rounded-xl border border-rose-200 bg-rose-50 px-3 py-2.5 text-sm font-bold text-rose-700">
   626→          <XCircle size={17} className="shrink-0" />
   627→          <span>{outcome.text}</span>
   628→        </div>
   629→      )}
   630→
   631→      {/* recently scanned trail (click = re-open) */}
   632→      {recent.length > 0 && (
   633→        <div className="mt-2.5 flex flex-wrap items-center gap-1.5">
   634→          <span className="text-[11px] font-semibold text-[#8A9884]">اسکن‌های اخیر:</span>
   635→          {recent.map((c) => (
   636→            <button
   637→              key={c}
   638→              type="button"
   639→              onClick={() => void submit(c)}
   640→              aria-label={`باز کردن مجدد ${c}`}
   641→              className="pz-barcode inline-flex min-h-[44px] items-center rounded-full border border-[#E4DCC8] bg-white px-3 py-1 text-xs font-bold tracking-wider text-[#4A5A44] transition-all hover:border-[#DAA520] hover:bg-[#FFFDF5] hover:text-[#8A6508] active:scale-[0.97]"
   642→            >
   643→              {c}
   644→            </button>
   645→          ))}
   646→        </div>
   647→      )}
   648→    </section>
   649→  )
   650→}
