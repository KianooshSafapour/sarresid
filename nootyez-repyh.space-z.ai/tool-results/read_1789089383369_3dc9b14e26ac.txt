    80→/**
    81→ * Normalize a scanned/typed order code for matching:
    82→ * strip Code39 start/stop `*` and all whitespace, map Persian/Arabic digits
    83→ * to ASCII (manual entry), uppercase. (Inline equivalent of sanitizeCode39
    84→ * from lib/code39 — that one keeps spaces, which scanners do send, and its
    85→ * charset filter would mangle the code shown in error messages.)
    86→ */
    87→function normalizeOrderCode(raw: string): string {
    88→  return String(raw ?? '')
    89→    .replace(/[\s*]+/g, '')
    90→    .replace(/[۰-۹]/g, (d) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d)))
    91→    .replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
    92→    .toUpperCase()
    93→}
    94→
    95→/**
    96→ * Stable-read dedupe for the camera detection loop: a raw read only counts
    97→ * when the SAME value is seen twice in a row (two consecutive detector hits)
    98→ * — filters one-off glitches/ghost reads. Pure so QA can eval it directly.
    99→ */
   100→function dedupeRead(last: string, raw: string): { accept: boolean; last: string } {
   101→  if (!raw) return { accept: false, last: '' }
   102→  return { accept: raw === last, last: raw }
   103→}
   104→
   105→/* QA/debug handles (same convention as window.__errs) — lets headless QA eval
   106→ * the scan normalization/dedupe logic without a camera. */
   107→if (typeof window !== 'undefined') {
   108→  ;(window as unknown as Record<string, unknown>).__hzScanDebug = { normalizeOrderCode, dedupeRead }
   109→}

   111→/** numeric input → number | null (accepts Persian digits too) */
   112→function num(v: string): number | null {
   113→  const t = String(v).trim().replace(/[۰-۹]/g, (d) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d)))
   114→  if (!t) return null
   115→  const n = Number(t)
   116→  return isFinite(n) ? n : null
   117→}
   118→
   119→const round1 = (n: number) => Math.round(n * 100) / 100

   121→function initRows(mode: SheetMode, order: DetailOrder): Record<number, RowState> {
   122→  const rec: Record<number, RowState> = {}
   123→  for (const it of order.items) {
   124→    rec[it.id] = {
   125→      qty: mode === 'receive'
   126→        ? (it.deliveredQty ?? it.qty)
   127→        : (it.confirmedQty ?? it.deliveredQty ?? it.qty),
   128→      printed: String(it.printedPrice ?? it.unitCost ?? 0),
   129→      final: String(it.finalCost ?? it.printedPrice ?? it.unitCost ?? 0),
   130→      status: it.status === 'MISSING' || it.status === 'REJECTED' ? it.status : 'OK',
   131→      note: it.note ?? '',
   132→    }
   133→  }
   134→  return rec
   135→}

   176→
   177→function DeliverySheet({ orderId, mode, user, onClose, onDone }: {
   178→  orderId: number | null
   179→  mode: SheetMode
   180→  user: PUser
   181→  onClose: () => void
   182→  onDone: () => void
   183→}) {
   184→  const [order, setOrder] = React.useState<DetailOrder | null>(null)
   185→  const [loading, setLoading] = React.useState(false)
   186→  const [rows, setRows] = React.useState<Record<number, RowState>>({})
   187→  const [scan, setScan] = React.useState('')
   188→  const [saving, setSaving] = React.useState(false)
   189→  const [barcodeFor, setBarcodeFor] = React.useState<number | null>(null)
   190→  const [barcodeVal, setBarcodeVal] = React.useState('')
   191→
   192→  React.useEffect(() => {
   193→    if (orderId == null) { setOrder(null); return }
   194→    let alive = true
   195→    setLoading(true)
   196→    setOrder(null)
   197→    setScan(''); setBarcodeFor(null); setBarcodeVal('')
   198→    api.get<DetailOrder>(`/api/orders/${orderId}`)
   199→      .then((o) => { if (!alive) return; setOrder(o); setRows(initRows(mode, o)) })
   200→      .catch(() => { if (!alive) return; toast.error('خطا در بارگذاری سفارش'); onClose() })
   201→      .finally(() => { if (alive) setLoading(false) })
   202→    return () => { alive = false }
   203→  }, [orderId, mode, onClose])
   204→
   205→  const upd = (id: number, patch: Partial<RowState>) =>
   206→    setRows((r) => (r[id] ? { ...r, [id]: { ...r[id], ...patch } } : r))
   207→
   208→  /* ---- barcode scan: mark item OK with full qty ---- */
   209→  const handleScan = () => {
   210→    const code = scan.trim()
   211→    if (!code || !order) return
   212→    const it = order.items.find((i) => (i.barcode ?? '') === code || (i.product?.barcode ?? '') === code)
   213→    if (!it) { toast.error('قلمی با این بارکد در سفارش یافت نشد'); return }
   214→    const q = mode === 'receive' ? it.qty : (it.deliveredQty ?? it.qty)
   215→    upd(it.id, { status: 'OK', qty: q, note: '' })
   216→    toast.success(`تایید شد ✓ — ${it.name}`)
   217→    setScan('')
   218→  }
   219→
   220→  /* ---- add barcode to an item (product) ---- */
   221→  const saveBarcode = async (itemId: number) => {
   222→    const barcode = barcodeVal.trim()
   223→    if (!barcode || orderId == null) return
   224→    try {
   225→      await api.patch(`/api/orders/${orderId}`, { action: 'addBarcode', userId: user.id, itemId, barcode })
   226→      toast.success('بارکد ذخیره شد ✓')
   227→      setBarcodeVal('')
   228→      setBarcodeFor(null)
   229→      setOrder((o) => (o
   230→        ? { ...o, items: o.items.map((i) => (i.id === itemId && !i.barcode ? { ...i, barcode } : i)) }
   231→        : o))
   232→    } catch (e) {
   233→      toast.error(e instanceof Error ? e.message : 'خطا در ذخیره بارکد')
   234→    }
   235→  }
   236→
   237→  /* ---- recalc footer ---- */
   238→  const totals = React.useMemo(() => {
   239→    if (!order) return { subtotal: 0, vat: 0, total: 0 }
   240→    let subtotal = 0
   241→    for (const it of order.items) {
   242→      const row = rows[it.id]
   243→      if (!row) continue
   244→      const cost = mode === 'receive'
   245→        ? (num(row.printed) ?? it.unitCost)
   246→        : (num(row.final) ?? it.printedPrice ?? it.unitCost)
   247→      subtotal += cost * row.qty
   248→    }
   249→    const vat = subtotal * VAT_RATE
   250→    return { subtotal, vat, total: subtotal + vat }
   251→  }, [order, rows, mode])
   252→
   253→  const delta = order ? totals.total - order.total : 0
   254→  const smallDelta = Math.abs(delta) <= DELTA_TOLERANCE
   255→
   256→  const submit = async () => {
   257→    if (!order) return
   258→    setSaving(true)
   259→    try {
   260→      const items = order.items.map((it) => {
   261→        const row = rows[it.id] ?? { qty: 0, printed: '', final: '', status: 'OK' as ItemStatus, note: '' }
   262→        return mode === 'receive'
   263→          ? {
   264→            id: it.id,
   265→            deliveredQty: row.qty,
   266→            printedPrice: num(row.printed),
   267→            status: row.status,
   268→            note: row.note || null,
   269→            finalCost: num(row.printed),
   270→          }
   271→          : {
   272→            id: it.id,
   273→            confirmedQty: row.qty,
   274→            finalCost: num(row.final),
   275→            status: row.status,
   276→            note: row.note || null,
   277→          }
   278→      })
   279→      await api.patch(`/api/orders/${order.id}`, {
   280→        action: mode === 'receive' ? 'receive' : 'confirm',
   281→        userId: user.id,
   282→        items,
   283→      })
   284→      toast.success(mode === 'receive' ? 'تحویل سفارش ثبت شد ✓' : 'تایید نهایی انبار ثبت شد ✓')
   285→      onDone()
   286→      onClose()
   287→    } catch (e) {
   288→      toast.error(e instanceof Error ? e.message : 'خطا در ثبت')
   289→    } finally {
   290→      setSaving(false)
   291→    }
   292→  }
   293→
   294→  const isToday = order ? order.receivingDate.slice(0, 10) === todayISO() : false
   295→
   296→  return (
   297→    <Modal
   298→      open={orderId != null}
   299→      onClose={onClose}
   300→      wide
   301→      title={mode === 'receive' ? 'دریافت تحویل سفارش' : 'تایید نهایی انبار'}
   302→    >
   303→      {loading || !order ? (
   304→        <Loading label="در حال بارگذاری اقلام سفارش…" />
   305→      ) : (
   306→        <div className="space-y-4">
   307→          {/* ---- header ---- */}
   308→          <div className="flex flex-wrap items-center justify-between gap-2 rounded-2xl border border-[#E4DCC8] bg-white p-4">
   309→            <div>
   310→              <div className="flex items-center gap-2">
   311→                <span className="text-lg font-extrabold text-[#253A2A]">{order.code}</span>
   312→                <StatusBadge status={order.status} />
   313→              </div>
   314→              <div className="mt-0.5 text-sm text-[#6B7A66]">تامین‌کننده: <b className="text-[#3E6B4A]">{order.supplier?.name ?? '—'}</b></div>
   315→            </div>
   316→            <div className="text-left">
   317→              <Badge className={isToday ? 'border-emerald-200 bg-emerald-50 text-emerald-700' : 'border-[#E4DCC8] bg-[#F5F2E8] text-[#4A5A44]'}>
   318→                {isToday ? 'تحویل صبح امروز' : `تحویل: ${fmtJalali(order.receivingDate)}`}
   319→              </Badge>
   320→              <div className="mt-1 text-xs text-[#8A9884]">
   321→                جمع اصلی سفارش: <b className="tabular-nums">{fmtMoney(order.total)}</b> · {toFaDigits(order.items.length)} قلم
   322→              </div>
   323→            </div>
   324→          </div>
   325→
   326→          {/* ---- scan box ---- */}
   327→          <div className="flex items-center gap-2 rounded-xl border border-[#93C572]/50 bg-[#F3F7EF] px-3 py-1.5">
   328→            <ScanLine size={18} className="shrink-0 text-[#3E6B4A]" />
   329→            <input
   330→              autoFocus
   331→              value={scan}
   332→              onChange={(e) => setScan(e.target.value)}
   333→              onKeyDown={(e) => { if (e.key === 'Enter') { e.preventDefault(); handleScan() } }}
   334→              placeholder="بارکد را اسکن کنید و Enter بزنید…"
   335→              dir="ltr"
   336→              className="h-11 w-full bg-transparent text-sm text-[#253A2A] outline-none placeholder:text-[#A8A28C]"
   337→              aria-label="اسکن بارکد"
   338→            />
   339→          </div>
   340→
   341→          {/* ---- item cards ---- */}
   342→          <div className="max-h-[46vh] space-y-3 overflow-y-auto pl-1 pz-scroll">
   343→            {order.items.map((it) => {
   344→              const row = rows[it.id]
   345→              if (!row) return null
   346→              const cost = mode === 'receive'
   347→                ? (num(row.printed) ?? it.unitCost)
   348→                : (num(row.final) ?? it.printedPrice ?? it.unitCost)
   349→              const line = cost * row.qty
   350→              return (
   351→                <Card key={it.id} className="p-3">
   352→                  {/* product head */}
   353→                  <div className="flex items-center gap-3">
   354→                    <ProductImage src={it.product?.imageUrl} name={it.name} size={48} />
   355→                    <div className="min-w-0 flex-1">
   356→                      <div className="truncate text-sm font-bold text-[#253A2A]">{it.name}</div>
   357→                      <div className="font-mono text-[11px] text-[#8A9884]" dir="ltr">{it.barcode || it.product?.barcode || '—'}</div>
   358→                      <div className="text-[11px] text-[#6B7A66]">
   359→                        مقدار سفارش: <b className="tabular-nums">{toFaDigits(it.qty)}</b>
   360→                        {mode === 'receive' && it.deliveredQty != null && <> · دریافتی قبلی: <b className="tabular-nums">{toFaDigits(it.deliveredQty)}</b></>}
   361→                      </div>
   362→                    </div>
   363→                    <div className="shrink-0 text-left">
   364→                      <div className="text-[10px] text-[#8A9884]">جمع خط</div>
   365→                      <div className="text-xs font-bold tabular-nums text-[#3E6B4A]">{fmtMoney(line)}</div>

   369→                  {/* qty + price */}
   370→                  <div className="mt-3 grid gap-3 sm:grid-cols-2">
   371→                    <QtyStepper
   372→                      label={mode === 'receive' ? 'تعداد دریافتی' : 'تعداد تایید نهایی'}
   373→                      value={row.qty}
   374→                      onChange={(n) => upd(it.id, { qty: n })}
   375→                    />
   376→                    {mode === 'receive' ? (
   377→                      <Field label="قیمت چاپ‌شده (تومان)" hint="قیمت چاپ‌شده روی بسته">
   378→                        <input
   379→                          type="number" inputMode="decimal" min={0} step="any"
   380→                          value={row.printed}
   381→                          onChange={(e) => upd(it.id, { printed: e.target.value })}
   382→                          className={inputCls}
   383→                        />
   384→                      </Field>
   385→                    ) : (
   386→                      <Field label="قیمت نهایی واحد (تومان)" hint="پیش‌فرض: قیمت چاپ‌شده / قیمت خرید">
   387→                        <input
   388→                          type="number" inputMode="decimal" min={0} step="any"
   389→                          value={row.final}
   390→                          onChange={(e) => upd(it.id, { final: e.target.value })}
   391→                          className={inputCls}
   392→                        />
   393→                      </Field>
   394→                    )}
   395→                  </div>
   396→
   397→                  {/* status toggles */}
   398→                  <div className="mt-3 grid grid-cols-3 gap-2">
   399→                    {STATUS_OPTS.map((o) => (
   400→                      <button
   401→                        key={o.v} type="button"

   409→                        {o.label}

   412→                  </div>
   413→
   414→                  {/* note when MISSING / REJECTED */}
   415→                  {row.status !== 'OK' && (
   416→                    <div className="mt-3">
   417→                      <Field label="توضیح" hint="دلیل ناقص یا مردود بودن را بنویسید">

   426→                  )}
   427→
   428→                  {/* add barcode */}
   429→                  <div className="mt-3">
   430→                    {barcodeFor === it.id ? (
   431→                      <div className="flex flex-col gap-2 sm:flex-row">

   507→function readRecentCodes(): string[] {
   508→  try {
   509→    const raw = window.sessionStorage.getItem(SCAN_RECENT_KEY)
   510→    const arr = raw ? JSON.parse(raw) : []
   511→    return Array.isArray(arr) ? arr.filter((c) => typeof c === 'string').slice(0, 3) : []
   512→  } catch {
   513→    return []
   514→  }
   515→}

   517→function writeRecentCodes(codes: string[]) {
   518→  try {
   519→    window.sessionStorage.setItem(SCAN_RECENT_KEY, JSON.stringify(codes.slice(0, 3)))
   520→  } catch {
   521→    /* private mode etc. */
   522→  }

   760→/* =============== camera scan modal (BarcodeDetector + graceful fallback) =============== */
   761→
   762→/**
   763→ * Full-screen camera viewfinder for order-barcode scanning.
   764→ * - Feature detection first: no BarcodeDetector / no getUserMedia → «unsupported»
   765→ *   fallback (browser can't do it). getUserMedia failure (denied / no device /
   766→ *   busy / insecure context) → «unavailable» fallback with a specific hint.
   767→ * - Live path: environment-facing stream in a premium viewfinder (gold corner
   768→ *   brackets, pz-cam-beam sweep), code_39 polled ~250 ms; a read is accepted
   769→ *   only when the same raw value appears twice in a row (dedupeRead) →
   770→ *   stop stream → onCode(rawValue) → the ScanBar runs its normal resolve().
   771→ * - All tracks stop on close, on tab-hide (visibilitychange) and on unmount.
   772→ */
   773→function ScanCameraModal({ open, onClose, onCode, onStatusChange }: {
   774→  open: boolean
   775→  onClose: () => void
   776→  onCode: (raw: string) => void
   777→  onStatusChange?: (status: CamStatus) => void
   778→}) {
   779→  const videoRef = React.useRef<HTMLVideoElement | null>(null)
   780→  const streamRef = React.useRef<MediaStream | null>(null)
   781→  const [status, setStatus] = React.useState<CamStatus>('starting')
   782→  const [errHint, setErrHint] = React.useState('')
   783→
   784→  const stopStream = React.useCallback(() => {
   785→    const st = streamRef.current
   786→    if (!st) return
   787→    for (const t of st.getTracks()) t.stop()
   788→    streamRef.current = null
   789→  }, [])
   790→
   791→  React.useEffect(() => {
   792→    if (!open) return
   793→    let cancelled = false
   794→    let timer: ReturnType<typeof setInterval> | null = null
   795→
   796→    const apply = (s: CamStatus) => {
   797→      setStatus(s)
   798→      onStatusChange?.(s)
   799→    }
   800→
   801→    /** first stable read → stop everything, hand the raw code to the bar */
   802→    const finish = (raw: string) => {
   803→      cancelled = true
   804→      if (timer) { clearInterval(timer); timer = null }
   805→      stopStream()
   806→      onCode(raw)
   807→    }
   808→
   809→    const start = async () => {
   810→      apply('starting')
   811→      setErrHint('')
   812→      const Ctor = window.BarcodeDetector
   813→      if (!Ctor || !navigator.mediaDevices?.getUserMedia) {
   814→        apply('unsupported')
   815→        return
   816→      }
   817→      try {
   818→        const stream = await navigator.mediaDevices.getUserMedia({
   819→          video: { facingMode: 'environment' },
   820→          audio: false,
   821→        })
   822→        if (cancelled) {
   823→          for (const t of stream.getTracks()) t.stop()
   824→          return
   825→        }
   826→        streamRef.current = stream
   827→        const v = videoRef.current
   828→        if (v) {
   829→          v.srcObject = stream
   830→          try { await v.play() } catch { /* muted+playsInline usually autoplay fine */ }
   831→        }
   832→        const detector = new Ctor({ formats: ['code_39'] })
   833→        apply('live')
   834→        let last = ''
   835→        timer = setInterval(() => {
   836→          if (cancelled) return
   837→          const vid = videoRef.current
   838→          if (!vid || vid.readyState < 2 || vid.videoWidth === 0) return
   839→          detector.detect(vid)
   840→            .then((hits) => {
   841→              if (cancelled) return
   842→              const raw = hits && hits.length > 0 && hits[0]?.rawValue ? String(hits[0].rawValue) : ''
   843→              if (!raw) return
   844→              const d = dedupeRead(last, raw)
   845→              last = d.last
   846→              if (d.accept) finish(raw)
   847→            })
   848→            .catch(() => { /* transient decode failure — keep polling */ })
   849→        }, 250)
   850→      } catch (e) {
   851→        if (cancelled) return
   852→        const name = e instanceof DOMException ? e.name : ''
   853→        if (name === 'NotAllowedError' || name === 'PermissionDeniedError' || name === 'SecurityError') {
   854→          setErrHint('دسترسی به دوربین رد شد. لطفاً از نوار آدرس مرورگر اجازه دوربین را برای این صفحه فعال کنید.')
   855→        } else if (name === 'NotFoundError' || name === 'DevicesNotFoundError' || name === 'OverconstrainedError') {
   856→          setErrHint('دوربینی روی این دستگاه پیدا نشد.')
   857→        } else if (name === 'NotReadableError' || name === 'TrackStartError' || name === 'AbortError') {
   858→          setErrHint('دوربین در دسترس نیست — احتمالاً برنامه دیگری آن را در حال استفاده است.')
   859→        } else {
   860→          setErrHint('باز کردن دوربین ممکن نشد — این قابلیت به محیط امن (HTTPS) نیاز دارد.')
   861→        }
   862→        apply('unavailable')
   863→      }
   864→    }
   865→
   866→    void start()
   867→
   868→    // extra safety: stop the camera when the tab is hidden or Escape pressed
   869→    const onKey = (ev: KeyboardEvent) => { if (ev.key === 'Escape') onClose() }
   870→    const onVis = () => { if (document.hidden) onClose() }
   871→    document.addEventListener('keydown', onKey)
   872→    document.addEventListener('visibilitychange', onVis)
   873→
   874→    return () => {
   875→      cancelled = true
   876→      if (timer) { clearInterval(timer); timer = null }
   877→      document.removeEventListener('keydown', onKey)
   878→      document.removeEventListener('visibilitychange', onVis)
   879→      stopStream()
   880→    }
   881→  }, [open, onClose, onCode, onStatusChange, stopStream])
   882→
   883→  if (!open) return null
   884→
   885→  const fallback = status === 'unsupported' || status === 'unavailable'
   886→
   887→  return (
   888→    <div
   889→      dir="rtl"
   890→      role="dialog"
   891→      aria-modal="true"
   892→      aria-label="اسکن بارکد سفارش با دوربین"
   893→      className="fixed inset-0 z-[60] flex flex-col bg-[#0E120C]/80 backdrop-blur-md"
   894→    >
   895→      {/* header */}
   896→      <div className="flex items-center justify-between gap-2 px-4 pt-4">
   897→        <div className="flex items-center gap-2 text-[#F3E7C4]">
   898→          <Camera size={18} />
   899→          <span className="text-sm font-bold">اسکن با دوربین</span>
   900→        </div>
   901→        <button
   902→          type="button"
