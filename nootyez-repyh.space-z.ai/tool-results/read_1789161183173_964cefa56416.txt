    45→/** Persian digits, grouped with «٬» — e.g. ۱۲٬۵۰۰٬۰۰۰ تومان */
    46→const faMoney = (n: number) => `${toFaDigits(new Intl.NumberFormat('en-US').format(Math.round(n)).replace(/,/g, '٬'))} تومان`

    50→const fa1 = (n: number) => toFaDigits((Math.round(n * 10) / 10).toFixed(1).replace('.', '٫'))
