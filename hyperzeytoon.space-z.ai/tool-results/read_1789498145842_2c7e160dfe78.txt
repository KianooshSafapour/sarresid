   342→// ================= Cheque calendar =================
   343→// ================= due heatmap (۸ هفته آینده) =================
   344→function DueHeatmap({
   345→  cheques, holidays, selected, onSelect,
   346→}: {
   347→  cheques: ChequeRow[]
   348→  holidays: Set<string>
   349→  selected: string | null
   350→  onSelect: (iso: string | null) => void
   351→}) {
   352→  const days = React.useMemo(() => {
   353→    const today = new Date()
   354→    today.setHours(0, 0, 0, 0)
   355→    const unsettled = cheques.filter((c) => !['COLLECTED', 'CLEARED', 'REJECTED', 'CANCELLED'].includes(c.status))
   356→    const byDay = new Map<string, { count: number; amount: number }>()
   357→    for (const c of unsettled) {
   358→      const k = isoDay(c.dueDate)
   359→      const e = byDay.get(k) ?? { count: 0, amount: 0 }
   360→      e.count++
   361→      e.amount += c.amount
   362→      byDay.set(k, e)
   363→    }
   364→    const out: { iso: string; g: Date; jd: number; jmName: string; isToday: boolean; isFriday: boolean; isHoliday: boolean; count: number; amount: number }[] = []
   365→    for (let i = 0; i < 56; i++) {
   366→      const g = new Date(today)
   367→      g.setDate(g.getDate() + i)
   368→      const j = toJalali(g)
   369→      const iso = isoDay(g)
   370→      const e = byDay.get(iso) ?? { count: 0, amount: 0 }
   371→      out.push({
   372→        iso, g, jd: j.jd, jmName: JALALI_MONTHS[j.jm - 1].slice(0, 3),
   373→        isToday: i === 0,
   374→        isFriday: g.getDay() === 5,
   375→        isHoliday: holidays.has(iso),
   376→        count: e.count, amount: e.amount,
   377→      })
   378→    }
   379→    return { days: out, max: Math.max(1, ...out.map((d) => d.amount)) }
   380→  }, [cheques, holidays])
   381→
   382→  const intensity = (amount: number) => {
   383→    if (!amount) return 'transparent'
   384→    const r = amount / days.max
   385→    if (r > 0.66) return '#3E7C59'
   386→    if (r > 0.33) return 'rgba(62,124,89,0.55)'
   387→    return 'rgba(62,124,89,0.22)'
   388→  }
   389→
   390→  // arrange into 7 weekday rows × 8 week columns (RTL: first week on the right)
   391→  const cells: (typeof days.days)[number][] = days.days
   392→  const weekCols: (typeof cells)[] = []

   480→    for (const c of cheques) {
   481→      if (['REJECTED'].includes(c.status)) continue
   482→      const k = isoDay(c.dueDate)
   483→      if (!m.has(k)) m.set(k, { count: 0, amount: 0, cheques: [] })
   484→      const e = m.get(k)!
   485→      e.count++
   486→      e.amount += c.amount
   487→      e.cheques.push(c)
   488→    }

   504→    let nm = jm + delta
   505→    let ny = jy
   506→    if (nm > 12) { nm = 1; ny++ }
   507→    if (nm < 1) { nm = 12; ny-- }

   509→    setJy(ny)
   510→    setSelected(null)

   740→// ================= split dialog =================
   741→function SplitChequeDialog({
   742→  open, onOpenChange, cheque, holidays, onDone,

   746→  cheque: ChequeRow
   747→  holidays: Set<string>

   757→    if (open) {

   760→        { amount: String(half), dueDate: null },
   761→        { amount: String(cheque.amount - half), dueDate: null },

   764→  }, [open, cheque])

   766→  const sum = parts.reduce((s, p) => s + (Number(p.amount) || 0), 0)

   773→        body: {
   774→          id: cheque.id,
   775→          parts: parts.map((p) => ({ amount: Number(p.amount), dueDate: p.dueDate ? p.dueDate.toISOString() : null })),

   802→                <div className="flex items-center justify-between">
   803→                  <p className="text-xs font-bold">بخش {toFaDigits(i + 1)}</p>

   805→                    <Button variant="ghost" size="sm" className="h-8 text-xs text-pomegranate" onClick={() => setParts((prev) => prev.filter((_, j) => j !== i))}>

   808→                  )}

   810→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
