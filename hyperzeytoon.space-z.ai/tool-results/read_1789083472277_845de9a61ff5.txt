   257→// ================= Start dialog =================
   258→function StartDialog({
   259→  open, onOpenChange, onStarted,

   261→  open: boolean
   262→  onOpenChange: (v: boolean) => void
   263→  onStarted: (id: string) => void

   270→  const [busy, setBusy] = React.useState(false)

   274→    api<{ categories: string[] }>('/api/products?limit=1')

   358→// ================= Detail dialog (the counting workspace) =================
   359→function DetailDialog({ id, onClose }: { id: string; onClose: () => void }) {
   360→  const { toast } = useToast()
   361→  const user = useApp((s) => s.user)
   362→  const setSection = useApp((s) => s.setSection)
   363→  const [data, setData] = React.useState<DetailData | null>(null)
   364→  const [loading, setLoading] = React.useState(true)
   365→  const [filter, setFilter] = React.useState<FilterKey>('all')
   366→  const [search, setSearch] = React.useState('')
   367→  const [edits, setEdits] = React.useState<Record<string, number>>({}) // productId -> staged qty
   368→  const [savingRows, setSavingRows] = React.useState<Set<string>>(new Set())
   369→  const [scanFlash, setScanFlash] = React.useState<string | null>(null)
   370→  const [commitOpen, setCommitOpen] = React.useState(false)
   371→  const [commitBusy, setCommitBusy] = React.useState(false)
   372→  const [wedgeBuf, setWedgeBuf] = React.useState('') // keyboard-wedge visual state
   373→  const wedgeRef = React.useRef({ buf: '', timer: null as ReturnType<typeof setTimeout> | null })
   374→  const rowRefs = React.useRef<Record<string, HTMLDivElement | null>>({})
   375→
   376→  const load = React.useCallback(async () => {
   377→    try {
   378→      const d = await api<DetailData>(`/api/stock-counts/${id}`)
   379→      setData(d)
   380→      setEdits({})
   381→    } catch (e) {
   382→      toast({ title: 'خطا در دریافت جلسه جرد', description: (e as Error).message, variant: 'destructive' })
   383→    } finally {
   384→      setLoading(false)
   385→    }
   386→  }, [id, toast])
   387→
   388→  React.useEffect(() => {
   389→    load()
   390→  }, [load])
   391→
   392→  const isInProgress = data?.session.status === 'IN_PROGRESS'
   393→
   394→  // ---------- stats derived live from items (survives optimistic updates) ----------
   395→  const stats = React.useMemo(() => {
   396→    const items = data?.items ?? []
   397→    const counted = items.filter((i) => i.countedQty !== null)
   398→    const diffs = counted.filter((i) => i.countedQty !== i.systemStock)
   399→    let shortageValue = 0
   400→    let surplusValue = 0
   401→    for (const i of diffs) {
   402→      const v = (i.countedQty! - i.systemStock) * i.buyPrice
   403→      if (v < 0) shortageValue += -v
   404→      else surplusValue += v
   405→    }
   406→    return {
   407→      totalItems: items.length,
   408→      countedItems: counted.length,
   409→      diffItems: diffs.length,
   410→      diffValue: Math.round(surplusValue - shortageValue),
   411→      shortageValue: Math.round(shortageValue),
   412→      surplusValue: Math.round(surplusValue),
   413→    }
   414→  }, [data?.items])
   415→
   416→  // ---------- save one row ----------
   417→  const saveRow = React.useCallback(
   418→    async (item: CountItem, qty: number) => {
   419→      if (!isInProgress) return
   420→      setSavingRows((s) => new Set(s).add(item.productId))
   421→      try {
   422→        await api(`/api/stock-counts/${id}`, {
   423→          method: 'POST',
   424→          body: { action: 'count', productId: item.productId, countedQty: qty },
   425→        })
   426→        setEdits((e) => {
   427→          const rest = { ...e }
   428→          delete rest[item.productId]
   429→          return rest
   430→        })
   431→        setData((d) => {
   432→          if (!d) return d
   433→          return {
   434→            ...d,
   435→            items: d.items.map((it) =>
   436→              it.productId === item.productId
   437→                ? { ...it, countedQty: qty, countedAt: new Date().toISOString(), countedByName: user?.name ?? '' }
   438→                : it
   439→            ),
   440→          }
   441→        })
   442→      } catch (e) {
   443→        toast({ title: 'خطا در ثبت شمارش', description: (e as Error).message, variant: 'destructive' })
   444→      } finally {
   445→        setSavingRows((s) => {
   446→          const n = new Set(s)
   447→          n.delete(item.productId)
   448→          return n
   449→        })
   450→      }
   451→    },
   452→    [id, isInProgress, toast, user?.name]
   453→  )
   454→
   455→  // ---------- barcode scan handler ----------
   456→  const handleScan = React.useCallback(
   457→    (code: string) => {
   458→      if (!data) return
   459→      const item = data.items.find((it) => it.barcodes.includes(code))
   460→      if (!item) {
   461→        toast({ title: 'بارکد شناخته نشد', description: `کد ${toFaDigits(code)} در محدوده این جلسه جرد نیست`, variant: 'destructive' })
   462→        return
   463→      }
   464→      setFilter('all')
   465→      setSearch('')
   466→      // scroll + flash + focus its qty input
   467→      requestAnimationFrame(() => {
   468→        rowRefs.current[item.productId]?.scrollIntoView({ behavior: 'smooth', block: 'center' })
   469→      })
   470→      setScanFlash(item.productId)
   471→      setTimeout(() => setScanFlash(null), 1800)
   472→      const input = rowRefs.current[item.productId]?.querySelector('input[data-qty]') as HTMLInputElement | null
   473→      input?.focus()
   474→      input?.select()
   475→    },
   476→    [data, toast]
   477→  )
   478→
   479→  // ---------- global keyboard-wedge capture (digits typed while focus is outside inputs) ----------
   480→  React.useEffect(() => {
   481→    if (!isInProgress) return
   482→    const onKey = (e: KeyboardEvent) => {
   483→      const t = e.target as HTMLElement | null
   484→      const inField = t && (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA' || t.tagName === 'SELECT' || t.isContentEditable)
   485→      if (inField) return
   486→      if (/^\d$/.test(e.key)) {
   487→        const w = wedgeRef.current
   488→        w.buf = (w.buf + e.key).slice(-32)
   489→        setWedgeBuf(w.buf)
