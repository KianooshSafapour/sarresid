   247→// ================= Cheque calendar =================
   248→function ChequeCalendar({
   249→  cheques, holidays,
   250→}: {
   251→  cheques: ChequeRow[]
   252→  holidays: Set<string>
   253→}) {
   254→  const now = React.useMemo(() => toJalali(new Date()), [])
   255→  const [jy, setJy] = React.useState(now.jy)
   256→  const [jm, setJm] = React.useState(now.jm)
   257→  const [selected, setSelected] = React.useState<string | null>(null)
   258→
   259→  const cells = React.useMemo(() => jalaliMonthGrid(jy, jm), [jy, jm])
   260→
   261→  const byDay = React.useMemo(() => {
   262→    const m = new Map<string, { count: number; amount: number; cheques: ChequeRow[] }>()
   263→    for (const c of cheques) {
   264→      if (['REJECTED'].includes(c.status)) continue
   265→      const k = isoDay(c.dueDate)
   266→      if (!m.has(k)) m.set(k, { count: 0, amount: 0, cheques: [] })
   267→      const e = m.get(k)!
   268→      e.count++
   269→      e.amount += c.amount
   270→      e.cheques.push(c)
   271→    }
   272→    return m
   273→  }, [cheques])
   274→
   275→  const monthTotal = React.useMemo(() => {
   276→    let sum = 0
   277→    let count = 0
   278→    for (const cell of cells) {
   279→      if (!cell) continue
   280→      const e = byDay.get(isoDay(cell.gDate))
   281→      if (e) { sum += e.amount; count += e.count }
   282→    }
   283→    return { sum, count }
   284→  }, [cells, byDay])
   285→
   286→  const move = (delta: number) => {
   287→    let nm = jm + delta
   288→    let ny = jy
   289→    if (nm > 12) { nm = 1; ny++ }
   290→    if (nm < 1) { nm = 12; ny-- }
   291→    setJm(nm)
   292→    setJy(ny)
   293→    setSelected(null)
   294→  }
   295→
   296→  const dayCheques = selected ? byDay.get(selected)?.cheques ?? [] : []
   297→
   298→  return (
   299→    <div className="rounded-2xl border border-border bg-card p-4">
   300→      <div className="flex items-center justify-between mb-3">
   301→        <Button variant="ghost" size="sm" className="h-9 w-9 p-0 touch-target" onClick={() => move(1)} aria-label="ماه بعد">‹</Button>
   302→        <div className="text-center">
   303→          <p className="font-bold text-sm">{JALALI_MONTHS[jm - 1]} {toFaDigits(jy)}</p>
   304→          <p className="text-[11px] text-muted-foreground num">{toFaDigits(monthTotal.count)} چک — {money(monthTotal.sum)} تومان</p>
   305→        </div>
   306→        <Button variant="ghost" size="sm" className="h-9 w-9 p-0 touch-target" onClick={() => move(-1)} aria-label="ماه قبل">›</Button>
   307→      </div>
   308→
   309→      <div className="grid grid-cols-7 gap-1 mb-1">
   310→        {JALALI_WEEKDAYS_SHORT.map((d, i) => (
   311→          <div key={i} className={cn('text-center text-[11px] font-medium py-1', i === 6 ? 'text-pomegranate' : 'text-muted-foreground')}>{d}</div>
   312→        ))}
   313→      </div>
   314→      <div className="grid grid-cols-7 gap-1">
   315→        {cells.map((cell, i) => {
   316→          if (!cell) return <div key={`e${i}`} />
   317→          const key = isoDay(cell.gDate)
   318→          const entry = byDay.get(key)
   319→          const isHoliday = holidays.has(key) || cell.isFriday
   320→          const isSel = selected === key
   321→          return (
   322→            <button
   323→              key={key}
   324→              type="button"
   325→              onClick={() => setSelected(isSel ? null : key)}
   326→              aria-label={`${formatJalaliFull(cell.gDate)}${entry ? ` — ${entry.count} چک` : ''}`}
   327→              className={cn(
   328→                'relative h-14 sm:h-16 rounded-xl border text-center transition-colors flex flex-col items-center justify-center gap-0.5',
   329→                isSel ? 'border-primary ring-2 ring-primary/30 bg-primary/5' : 'border-border hover:bg-accent/60',
   330→                isHoliday && 'bg-pomegranate/5'
   331→              )}
   332→            >
   333→              <span className={cn('text-xs num font-bold', isHoliday ? 'text-pomegranate' : 'text-foreground')}>{toFaDigits(cell.jd)}</span>
   334→              {entry && (
   335→                <>
   336→                  <span className="text-[10px] num font-bold text-[#8a6f3c]">{toFaDigits(entry.count)} چک</span>
   337→                  <span className="absolute bottom-1 flex gap-0.5" aria-hidden>
   338→                    <span className="h-1.5 w-1.5 rounded-full bg-[#C9A227]" />
   339→                    {entry.amount > 500_000_000 && <span className="h-1.5 w-1.5 rounded-full bg-[#8A6F3C]" />}
   340→                  </span>
   341→                </>
   342→              )}
   343→              {isHoliday && !entry && <span className="absolute bottom-1 h-1 w-1 rounded-full bg-pomegranate/60" aria-hidden />}
   344→            </button>
   345→          )
   346→        })}
   347→      </div>
   348→
   349→      {selected && (
   350→        <div className="mt-3 pt-3 border-t border-border space-y-2 animate-in fade-in">
   351→          <p className="text-xs font-bold">{formatJalaliFull(isoToDate(selected))}</p>
   352→          {dayCheques.length === 0 ? (
   353→            <p className="text-xs text-muted-foreground">چکی در این روز سررسید نمی‌شود</p>
   354→          ) : (
   355→            dayCheques.map((c) => {
   356→              const ci = chequeStatusInfo(c.status)
   357→              return (
   358→                <div key={c.id} className="flex flex-wrap items-center gap-2 text-xs rounded-lg border border-border px-2.5 py-2">
   359→                  <span className="num font-bold">{c.number}</span>
   360→                  <span>{c.payeeName}</span>
   361→                  <span className="num font-bold text-primary ms-auto">{money(c.amount)} تومان</span>
   362→                  <StatusBadge label={ci.label} color={ci.color} />
   363→                </div>
   364→              )
   365→            })
   366→          )}
   367→        </div>
   368→      )}
   369→    </div>
   370→  )
   371→}

   373→// helper: isoDay (gregorian yyyy-mm-dd) back to Date
   374→function isoToDate(selected: string): Date {
   375→  const [y, m, d] = toEnDigits(selected).split('-').map(Number)
   376→  return new Date(y, m - 1, d)
   377→}

   405→
   406→  return (
   407→    <div className={cn('rounded-2xl border bg-card p-4 space-y-2.5', ['PENDING_OWNER'].includes(c.status) ? 'border-[#C9A227]/50' : 'border-border')}>

   422→        {c.collectedAt && <span>وصول: <span className="num">{formatJalali(c.collectedAt)}</span></span>}
   423→      </div>
   424→      {c.note && <p className="text-[11px] text-muted-foreground leading-5">{c.note}</p>}
   425→
   426→      <div className="flex flex-wrap gap-2 pt-1">
   427→        {c.status === 'PENDING_OWNER' && isOwner && (
   428→          <>
   429→            <Button size="sm" className="touch-target font-bold" disabled={busy} onClick={() => run(async () => { await onPatch(c.id, 'SIGNED', {}, 'چک امضا و تأیید شد ✍️') })}>
   430→              <PenLine className="h-4 w-4" /> امضا و تأیید
   431→            </Button>
   432→            <Button size="sm" variant="outline" className="touch-target text-pomegranate border-pomegranate/40" disabled={busy} onClick={() => setRejectOpen(true)}>
   433→              <XCircle className="h-4 w-4" /> رد

   435→          </>

   521→function SplitChequeDialog({
   522→  open, onOpenChange, cheque, holidays, onDone,
   523→}: {
   524→  open: boolean
   525→  onOpenChange: (v: boolean) => void
   526→  cheque: ChequeRow
   527→  holidays: Set<string>
   528→  onDone: () => void
   529→}) {
   530→  const [parts, setParts] = React.useState<{ amount: string; dueDate: Date | null }[]>([
   531→    { amount: '', dueDate: null },
   532→    { amount: '', dueDate: null },
   533→  ])
   534→  const [saving, setSaving] = React.useState(false)
   535→
   536→  React.useEffect(() => {
   537→    if (open) {
   538→      const half = Math.floor(cheque.amount / 2)
   539→      setParts([
   540→        { amount: String(half), dueDate: null },
   541→        { amount: String(cheque.amount - half), dueDate: null },
   542→      ])
   543→    }
   544→  }, [open, cheque])
   545→
   546→  const sum = parts.reduce((s, p) => s + (Number(p.amount) || 0), 0)
   547→  const diff = sum - cheque.amount
   548→
   549→  const submit = async () => {
   550→    setSaving(true)
   551→    try {
   552→      await api('/api/cheques/split', {
   553→        body: {
   554→          id: cheque.id,
   555→          parts: parts.map((p) => ({ amount: Number(p.amount), dueDate: p.dueDate ? p.dueDate.toISOString() : null })),
   556→        },
   557→      })
   558→      toast({ title: `چک به ${toFaDigits(parts.length)} بخش تقسیم شد ✂️`, description: 'بخش‌های جدید در انتظار امضای مالک هستند' })
   559→      onOpenChange(false)
   560→      onDone()
   561→    } catch (e) {
   562→      toast({ title: 'خطا در تقسیم چک', description: e instanceof Error ? e.message : undefined, variant: 'destructive' })
   563→    } finally {
   564→      setSaving(false)
   565→    }
   566→  }
   567→
   568→  return (
   569→    <Dialog open={open} onOpenChange={onOpenChange}>
   570→      <DialogContent className="w-[95vw] max-w-lg max-h-[90vh] overflow-y-auto">
   571→        <DialogHeader>
   572→          <DialogTitle className="text-base">تقسیم چک {cheque.number}</DialogTitle>
   573→          <DialogDescription className="text-xs">
   574→            مبلغ اصلی: <span className="num font-bold">{money(cheque.amount)}</span> تومان — جمع بخش‌ها باید دقیقاً برابر باشد.
   575→          </DialogDescription>
   576→        </DialogHeader>
   577→        <div className="space-y-2.5">
   578→          {parts.map((p, i) => {
   579→            const closed = p.dueDate ? isClosedDay(p.dueDate, holidays) : false
   580→            return (
   581→              <div key={i} className="rounded-xl border border-border p-3 space-y-2">
   582→                <div className="flex items-center justify-between">
   583→                  <p className="text-xs font-bold">بخش {toFaDigits(i + 1)}</p>
   584→                  {parts.length > 2 && (
   585→                    <Button variant="ghost" size="sm" className="h-8 text-xs text-pomegranate" onClick={() => setParts((prev) => prev.filter((_, j) => j !== i))}>
   586→                      حذف
   587→                    </Button>
   588→                  )}
   589→                </div>
   590→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   591→                  <Input
   592→                    value={p.amount}
   593→                    onChange={(e) => setParts((prev) => prev.map((x, j) => (j === i ? { ...x, amount: e.target.value.replace(/[^\d.]/g, '') } : x)))}
   594→                    inputMode="numeric"
   595→                    placeholder="مبلغ (تومان)"
   596→                    className="h-11 num"
   597→                    aria-label={`مبلغ بخش ${i + 1}`}
   598→                  />
   599→                  <JalaliDatePicker
   600→                    value={p.dueDate}
   601→                    onChange={(d) => setParts((prev) => prev.map((x, j) => (j === i ? { ...x, dueDate: d } : x)))}
   602→                    holidays={holidays}
   603→                    allowClear={false}
   604→                  />
   605→                </div>
   606→                {closed && (
   607→                  <div className="flex flex-wrap items-center gap-2 rounded-lg bg-[#C9A227]/10 border border-[#C9A227]/40 px-2.5 py-2 text-[11px] text-[#8a6f3c]" role="alert">
   608→                    <span className="font-bold">سررسید به تعطیلات می‌افتد؛ زودتر انتخاب کنید</span>
   609→                    {p.dueDate && (
   610→                      <Button
   611→                        type="button"
   612→                        variant="outline"
   613→                        size="sm"
   614→                        className="h-8 text-[11px]"
   615→                        onClick={() => setParts((prev) => prev.map((x, j) => (j === i ? { ...x, dueDate: suggestOpenDay(p.dueDate!, holidays) } : x)))}
   616→                      >
   617→                        پیشنهاد: {formatJalali(suggestOpenDay(p.dueDate, holidays))}
   618→                      </Button>
   619→                    )}
   620→                  </div>
   621→                )}
   622→              </div>
   623→            )
   624→          })}
   625→        </div>
   626→        <div className="flex items-center justify-between text-xs">
   627→          <Button
   628→            variant="outline"
   629→            size="sm"
   630→            className="touch-target"
   631→            onClick={() => setParts((prev) => [...prev, { amount: '', dueDate: null }])}
   632→            disabled={parts.length >= 6}
   633→          >
   634→            <Plus className="h-4 w-4" /> افزودن بخش
   635→          </Button>
   636→          <span className={cn('num font-bold', Math.abs(diff) <= 1 ? 'text-primary' : 'text-pomegranate')}>
   637→            جمع: {money(sum)} / {money(cheque.amount)} {Math.abs(diff) > 1 && `(اختلاف ${money(diff)})`}
   638→          </span>
   639→        </div>
   640→        <div className="flex justify-end gap-2">
   641→          <Button variant="outline" size="sm" className="touch-target" onClick={() => onOpenChange(false)} disabled={saving}>انصراف</Button>
   642→          <Button size="sm" className="touch-target font-bold" onClick={submit} disabled={saving || Math.abs(diff) > 1 || parts.some((p) => !Number(p.amount) || !p.dueDate)}>
   643→            {saving && <Loader2 className="h-4 w-4 animate-spin" />} تقسیم قطعی
   644→          </Button>
   645→        </div>
   646→      </DialogContent>
   647→    </Dialog>
   648→  )
   649→}
