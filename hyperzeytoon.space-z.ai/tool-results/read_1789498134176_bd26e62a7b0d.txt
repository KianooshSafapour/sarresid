   305→                <div className="grid grid-cols-2 gap-2">
   306→                  <Button
   307→                    variant="outline" className="gap-1.5 border-saffron/50 text-saffron"

   352→function statusInfo(status: string) {
   353→  return TASK_STATUSES.find((s) => s.key === status) ?? { label: status, color: '#8A8F98' }

   356→function TaskCard({
   357→  task, mine, busy, onOpen, onStatus, onEdit,
   358→}: {
   359→  task: TaskDTO
   360→  mine?: boolean
   361→  busy?: boolean
   362→  onOpen: () => void
   363→  onStatus?: (s: string) => void
   364→  onEdit?: () => void
   365→}) {
   366→  const pr = PRIORITIES.find((p) => p.key === task.priority) ?? PRIORITIES[1]
   367→  const st = statusInfo(task.status)
   368→  const overdue = task.dueDate && task.status !== 'DONE' && new Date(task.dueDate).getTime() < Date.now()
   369→  return (
   370→    <Card className="glow-border-static overflow-hidden">
   371→      <CardContent className="p-4 space-y-3">
   372→        <button onClick={onOpen} className="w-full text-right space-y-2 focus-visible:ring-2 focus-visible:ring-ring rounded-lg">
   373→          <div className="flex items-start justify-between gap-2">
   374→            <p className="font-bold leading-6">{task.title}</p>
   375→            <StatusBadge label={st.label} color={st.color} />
   376→          </div>
   377→          <div className="flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   378→            <span className="inline-flex items-center gap-1">
   379→              <span className="h-2 w-2 rounded-full" style={{ backgroundColor: pr.color }} />
   380→              {pr.label}
   381→            </span>
   382→            <Badge variant="secondary" className="rounded-full">{task.category}</Badge>
   383→            {task.fromOwner && <Badge className="rounded-full text-white" style={{ background: 'linear-gradient(120deg,#c9a227,#8a6f3c)' }}>👑 مالک</Badge>}
   384→            {task.dueDate && (
   385→              <span className={overdue ? 'text-pomegranate font-bold' : ''}>
   386→                📅 {formatJalali(task.dueDate)} {overdue && '(گذشته)'}
   387→              </span>
   388→            )}
   389→          </div>
   390→          {!mine && task.assigneeName && (
   391→            <div className="flex items-center gap-1.5 text-xs text-muted-foreground">
   392→              <UserAvatar name={task.assigneeName} color={task.assigneeColor ?? '#3E7C59'} size={20} />
   393→              {task.assigneeName}
   394→            </div>
   395→          )}
   396→          {task.status === 'PAUSED' && task.pauseReason && (
   397→            <p className="text-xs text-saffron flex items-center gap-1">
   398→              <PauseCircle className="h-3.5 w-3.5" /> {task.pauseReason}
   399→            </p>
   400→          )}
   401→        </button>
   402→        {mine && onStatus && task.status !== 'DONE' ? (
   403→          <div className="grid grid-cols-2 gap-2">
   404→            {task.status === 'TODO' && (
   405→              <Button variant="outline" size="sm" className="gap-1.5 border-saffron/50 text-saffron h-10" disabled={busy} onClick={() => onStatus('IN_PROGRESS')}>
   406→                <PlayCircle className="h-4 w-4" /> شروع
   407→              </Button>
   408→            )}
   409→            {task.status !== 'FOLLOW_UP' && (
   410→              <Button variant="outline" size="sm" className="h-10" disabled={busy} onClick={() => onStatus('FOLLOW_UP')}>
   411→                <Eye className="h-4 w-4" /> پیگیری
   412→              </Button>
   413→            )}
   414→            {task.status !== 'PAUSED' && (
   415→              <Button variant="outline" size="sm" className="gap-1.5 border-pomegranate/50 text-pomegranate h-10" onClick={() => onStatus('PAUSED')}>
   416→                <PauseCircle className="h-4 w-4" /> توقف
   417→              </Button>
   418→            )}
   419→            <Button size="sm" className="gap-1.5 h-10" disabled={busy} onClick={() => onStatus('DONE')}>
   420→              <CheckCircle2 className="h-4 w-4" /> انجام شد ✅
   421→            </Button>
   422→          </div>
   423→        ) : (
   424→          <div className="flex items-center justify-between gap-2">
   425→            <Button variant="outline" size="sm" className="gap-1.5 h-9" onClick={onOpen}>
   426→              جزئیات و گفتگو
   427→            </Button>
   428→            {onEdit && (
   429→              <Button variant="ghost" size="sm" className="h-9 gap-1.5" onClick={onEdit}>
   430→                <Pencil className="h-3.5 w-3.5" /> ویرایش
   431→              </Button>
   432→            )}
   433→          </div>
   434→        )}
   435→      </CardContent>
   436→    </Card>
   437→  )
   438→}

   440→function TaskEditor({
   441→  open, task, users, onClose, onSaved,
   442→}: {
   443→  open: boolean
   444→  task: TaskDTO | null
   445→  users: UserLite[]
   446→  onClose: () => void
   447→  onSaved: () => void
   448→}) {
   449→  const { toast } = useToast()
   450→  const [title, setTitle] = React.useState('')
   451→  const [description, setDescription] = React.useState('')
   452→  const [category, setCategory] = React.useState('عمومی')
   453→  const [priority, setPriority] = React.useState<string>('MEDIUM')
   454→  const [assignedToId, setAssignedToId] = React.useState<string>('')
   455→  const [dueDate, setDueDate] = React.useState<Date | null>(null)
   456→  const [saving, setSaving] = React.useState(false)
   457→
   458→  React.useEffect(() => {
   459→    if (open) {
   460→      setTitle(task?.title ?? '')
   461→      setDescription(task?.description ?? '')
   462→      setCategory(task?.category ?? 'عمومی')
   463→      setPriority(task?.priority ?? 'MEDIUM')
   464→      setAssignedToId(task?.assignedToId ?? '')
   465→      setDueDate(task?.dueDate ? new Date(task.dueDate) : null)
   466→    }
   467→  }, [open, task])
   468→
   469→  const save = async () => {
   470→    if (!title.trim()) return
   471→    setSaving(true)
   472→    try {

   481→        await api('/api/tasks', { body })

   557→// ---------- daily checklist (واقعی‌ترین دوست شیفت!) ----------
   558→interface ChecklistDTO {
   559→  id: string
   560→  title: string
   561→  items: { text: string }[]
   562→  state: { text: string; done: boolean }[]
   563→  completedAt?: string | null
   564→}
   565→
   566→export function DailyChecklist() {
   567→  const { toast } = useToast()
   568→  const [lists, setLists] = React.useState<ChecklistDTO[] | null>(null)
   569→  const [open, setOpen] = React.useState(true)
   570→  const day = jalaliKey(new Date())
   571→  const busyRef = React.useRef(false)
   572→
   573→  const load = React.useCallback(async () => {
   574→    try {
   575→      const d = await api<{ checklists: ChecklistDTO[] }>(`/api/checklists?day=${encodeURIComponent(day)}`)
   576→      setLists(d.checklists)
   577→    } catch {
   578→      setLists([])
   579→    }
   580→  }, [day])
   581→
   582→  React.useEffect(() => { load() }, [load])
   583→
   584→  const toggle = async (list: ChecklistDTO, idx: number) => {
   585→    if (busyRef.current) return
   586→    const state = (list.state.length === list.items.length && list.state.length > 0
   587→      ? list.state
   588→      : list.items.map((i) => ({ text: i.text, done: false }))
   589→    ).map((s, j) => (j === idx ? { ...s, done: !s.done } : s))
   590→    // optimistic
   591→    setLists((prev) => prev?.map((l) => (l.id === list.id ? { ...l, state, completedAt: null } : l)) ?? prev)
   592→    busyRef.current = true
   593→    try {
   594→      const d = await api<{ success: boolean; completedAt: string | null }>('/api/checklists', {
   595→        method: 'PATCH',
   596→        body: { id: list.id, day, state },
   597→      })
   598→      if (d.completedAt) {
   599→        toast({ title: 'آفرین! چک‌لیست امروز کامل شد +۲ امتیاز ⭐', description: 'شیفت تمیزی داشتی — دستت درد نکند 🌿' })
   600→      }
   601→      await load()
   602→    } catch {
   603→      load()
   604→    } finally {
   605→      busyRef.current = false
   606→    }
   607→  }
   608→
   609→  if (!lists) return null
   610→  if (lists.length === 0) return null
   611→
   612→  const allDone = lists.every((l) => l.completedAt)
   613→
   614→  return (
   615→    <Card className="glow-border-static overflow-hidden mb-4">
   616→      <button
   617→        type="button"
   618→        onClick={() => setOpen((o) => !o)}
   619→        className="w-full flex items-center justify-between gap-2 p-4 pb-3 text-right"
   620→        aria-expanded={open}
   621→      >
   622→        <span className="flex items-center gap-2 text-sm font-bold">
   623→          <ClipboardCheck className="h-4 w-4 text-primary" />
   624→          چک‌لیست امروز
   625→          {allDone ? (
   626→            <span className="rounded-full bg-primary/10 text-primary px-2.5 py-1 text-[11px] font-bold">کامل شد ✅ +۲</span>
   627→          ) : (
   628→            <span className="rounded-full bg-gold/15 text-gold px-2.5 py-1 text-[11px] font-bold num">
   629→              {toFaDigits(lists.reduce((s, l) => s + l.state.filter((x) => x.done).length, 0))}/{toFaDigits(lists.reduce((s, l) => s + l.items.length, 0))}
   630→            </span>
   631→          )}
   632→        </span>
   633→        <ChevronDown className={`h-4 w-4 text-muted-foreground transition-transform ${open ? 'rotate-180' : ''}`} />
   634→      </button>
   635→      {open && (
   636→        <CardContent className="pt-0 space-y-3">
   637→          {lists.map((l) => {
   638→            const state = l.state.length === l.items.length && l.state.length > 0
   639→              ? l.state
   640→              : l.items.map((i) => ({ text: i.text, done: false }))
   641→            return (
   642→              <div key={l.id} className="rounded-xl border p-3 space-y-2">
   643→                <p className="text-xs font-bold flex items-center gap-1.5">
   644→                  {l.title}
   645→                  {l.completedAt && <CheckCircle2 className="h-3.5 w-3.5 text-primary" />}
   646→                </p>
   647→                {l.items.map((item, idx) => {
   648→                  const done = state[idx]?.done ?? false
   649→                  return (
   650→                    <button
   651→                      key={idx}
   652→                      type="button"
   653→                      onClick={() => toggle(l, idx)}
   654→                      className={`w-full flex items-center gap-2.5 rounded-lg px-2.5 py-2.5 text-right text-xs transition-colors min-h-11 ${
   655→                        done ? 'bg-primary/10 text-primary line-through decoration-primary/40' : 'hover:bg-accent'
   656→                      }`}
   657→                      aria-pressed={done}
   658→                    >
   659→                      <span
   660→                        className={`h-5 w-5 rounded-md border flex items-center justify-center shrink-0 transition-colors ${
   661→                          done ? 'bg-primary border-primary text-primary-foreground' : 'border-input'
   662→                        }`}
   663→                      >
   664→                        {done && <CheckCircle2 className="h-3.5 w-3.5" />}
   665→                      </span>
   666→                      <span className="flex-1">{item.text}</span>
   667→                    </button>
   668→                  )
   669→                })}
   670→              </div>
   671→            )
   672→          })}
   673→          <p className="text-[10px] text-muted-foreground text-center">تکمیل همه‌ی موارد روزانه = ۲ امتیاز تشویقی ⭐</p>
   674→        </CardContent>
   675→      )}
   676→    </Card>
   677→  )
   678→}
