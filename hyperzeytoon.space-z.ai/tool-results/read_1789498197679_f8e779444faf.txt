    91→const CARD_COLORS = ['#3E7C59', '#C9A227', '#B07D2B', '#B33A3A', '#8A6F3C', '#7D5BA6']
    92→function colorOf(name: string) {
    93→  let h = 0
    94→  for (let i = 0; i < name.length; i++) h = (h * 31 + name.charCodeAt(i)) >>> 0
    95→  return CARD_COLORS[h % CARD_COLORS.length]
    96→}
