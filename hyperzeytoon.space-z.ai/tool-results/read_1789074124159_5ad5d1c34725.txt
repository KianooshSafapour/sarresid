   388→   388→function ProductsStep({
   389→   389→  catalog, provider, search, onSearch, category, categories, onCategory, cart, bump, setQty, setPrice,
   390→   390→}: {
   391→   391→  catalog: CatalogBundle | null
   392→   392→  provider: CatalogBundle['providers'][number] | null
   393→   393→  search: string
   394→   394→  onSearch: (v: string) => void
   395→   395→  category: string | null
   396→   396→  categories: string[]
   397→   397→  onCategory: (v: string | null) => void
   398→   398→  cart: Record<string, CartLine>
   399→   399→  bump: string | null
   400→   400→  setQty: (p: CatalogProduct, qty: number) => void
   401→   401→  setPrice: (p: CatalogProduct, price: number) => void
   402→   402→}) {
   403→   403→  const products = catalog?.products ?? []
   404→   404→  const list = React.useMemo(() => {
   405→   405→    let l = products
   406→   406→    if (search.trim()) {
   407→   407→      const q = search.trim()
   408→   408→      l = l.filter((p) => p.name.includes(q) || (p.altName ?? '').includes(q) || (p.brand ?? '').includes(q) || p.barcodes.some((b) => b.code.includes(q)))
   409→   409→    }
   410→   410→    if (category) l = l.filter((p) => p.category === category)
   411→   411→    if (provider) {
   412→   412→      const compNames = provider.companies.map((c) => c.name)
   413→   413→      if (compNames.length > 0) {
   414→   414→        l = [...l].sort((a, b) => {
   415→   415→          const aM = compNames.some((n) => (a.brand ?? '').includes(n) || a.name.includes(n)) ? 0 : 1
   416→   416→          const bM = compNames.some((n) => (b.brand ?? '').includes(n) || b.name.includes(n)) ? 0 : 1
   417→   417→          return aM - bM
